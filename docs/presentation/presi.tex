\begin{frame}{Simplices}
\begin{block}{Definition ($n$-simplex)}
Points $v_0,\dots,v_n \in \R^m$ are \alert{affinely independent} if $v_1 - v_0, \dots, v_n - v_0$ are linearly independent. If they are, their convex hull
\[
[v_0,\dots,v_n] = \Big\{ \textstyle\sum_{i=0}^n t_i v_i \;:\; t_i \ge 0,\ \sum_i t_i = 1 \Big\}
\]
is an \alert{$n$-simplex} with \emph{vertices} $v_i$. The simplex spanned by any nonempty subset of the vertices is a \alert{face}.
\src{hatcher2001}
\end{block}

\begin{frame}{Simplicial complexes}
\begin{block}{Definition (simplicial complex)}
A \alert{simplicial complex} $K$ is a finite collection of simplices such that
\begin{enumerate}
  \item every face of a simplex of $K$ is in $K$;
  \item any two simplices of $K$ are disjoint or intersect in a common face.
\end{enumerate}
Its \alert{vertex set} is the set of its 0-simplices.
\src{edelsbrunnerletscherzomorodian2002}
\end{block}
\begin{block}{Abstract version}
An \alert{abstract simplicial complex} on a vertex set $V$ is a collection $S$ of finite nonempty subsets of $V$ with $\{v\} \in S$ for all $v$, and $\tau \subseteq \sigma \in S \Rightarrow \tau \in S$. A $k$-simplex is a set of $k+1$ vertices. Placing the vertices in general position and filling in each simplex gives a geometric complex; the union of its simplices, as a space, is the \alert{geometric realization} $|S|$.
\src{zomorodiancarlsson2005}
\end{block}
\end{frame}

\begin{frame}{Graphs are simplicial complexes}
\note[item]{Simple: no edge from a vertex to itself and no repeated edges. An abstract complex cannot hold either, so a multigraph must be subdivided first.}
\begin{block}{Definition ($k$-skeleton)}
The \alert{$k$-skeleton} $K^{(k)}$ of a simplicial complex $K$ is the subcomplex of all simplices of dimension $\le k$; the 0-skeleton is the vertex set.
\src{edelsbrunnerharerpatel2008}
\end{block}
By the abstract definition, a finite \alert{simple} graph $G = (V, E)$ \emph{is} a simplicial complex of dimension $\le 1$; its realization $|G|$ is a \alert{topological graph}.\par

\begin{frame}{The nerve of a cover}
\begin{block}{Definition (nerve)}
Let $\mathcal U = \{U_\alpha\}_{\alpha \in A}$ be a finite cover of a space $X$ by nonempty sets. The \alert{nerve} $N(\mathcal U)$ is the abstract simplicial complex with vertex set $A$, where a set $\{\alpha_0,\dots,\alpha_k\} \subseteq A$ spans a $k$-simplex if and only if
\[ U_{\alpha_0} \cap \cdots \cap U_{\alpha_k} \ne \emptyset. \]
\src{singh2007mapper,hatcher2001}
\end{block}

\begin{frame}{The Nerve Theorem}
\begin{block}{Definitions}
\small Spaces $X \simeq Y$ are \alert{homotopy equivalent} if there are maps $X \to Y$ and $Y \to X$ whose composites are homotopic to the identities (backup); \emph{contractible} means $\simeq$ a point (a disk is, a circle is not).
$|N(\mathcal C)|$ is the \alert{geometric realization} of the nerve: the abstract complex drawn as a space, like the triangles on the previous slide.
\src{hatcher2001}
\end{block}
\begin{block}{Theorem (Nerve Theorem)}
If $X \subseteq \R^d$ and $\mathcal U$ is an open cover of $X$ in which every nonempty finite intersection of sets is contractible, then $X \simeq |N(\mathcal U)|$.
\par\vspace{2pt}\hfill{\scriptsize\textcolor{gray}{\parencite[Cor.~4G.3]{hatcher2001}, stated for paracompact spaces, which include every subset of $\R^d$ \parencite[Thm.~41.4]{munkres2000topology}}}
\end{block}
\begin{itemize}\footnotesize
  \item The circle's arcs are open, so this covers the previous slide.
  \item Closed convex sets also work (convex intersections). \hfill\parencite[Thm.~3.9]{bauerkerberrollrolle2023}
  \item \alert{Warning for later:} covers in data are rarely this nice, so a nerve need not match the space.
\end{itemize}
\end{frame}

\begin{frame}{Homology and Betti numbers}
\begin{block}{Definition ($\Z_2$ homology of a complex $K$)}
$C_k$ is the $\Z_2$-vector space with basis the $k$-simplices of $K$. The \alert{boundary map} $\partial_k : C_k \to C_{k-1}$ sends each simplex to the sum of its faces, e.g.\ $\partial_2\{a,b,c\} = \{b,c\} + \{a,c\} + \{a,b\}$. Then $\partial_k \partial_{k+1} = 0$, $H_k = \ker \partial_k / \operatorname{im} \partial_{k+1}$, and the \alert{Betti number} is $b_k = \dim H_k$.
\src{edelsbrunnerletscherzomorodian2002,hatcher2001}
\end{block}
\begin{itemize}\small
  \item $b_0$ = number of components; $b_1$ counts independent \alert{loops}. For a graph, $b_1$ is the \alert{cycle rank}; a loop here is never a self-edge.
  \item Rank--nullity gives $b_1 = E - V + C - \rank \partial_2$. A graph has no triangles, so $b_1(G) = E - V + C$.
\end{itemize}

\begin{tikzpicture}[scale=0.6, v/.style={circle, fill=black, inner sep=1.4pt}, lab/.style={font=\small}]
  \draw[thick] (0,0) node[v]{} node[lab,below left]{$a$} -- (1.6,0) node[v]{} node[lab,below right]{$b$} -- (0.8,1.3) node[v]{} node[lab,above]{$c$} -- cycle;

\end{frame}


\begin{frame}{Mapper: from spaces to data}
\begin{block}{Definition (Mapper)}
\footnotesize Let $f : X \to Z$ be continuous and $\mathcal U = \{U_\alpha\}$ a finite open cover of $Z$. Split each preimage $f^{-1}(U_\alpha)$ into its \alert{path components}. These form the \alert{pullback cover} $f^*\mathcal U$ of $X$, and the \alert{Mapper} is its nerve, $M(\mathcal U, f) = N(f^*\mathcal U)$.
\src{singh2007mapper,deymemoliwang2017}
\end{block}
\begin{block}{On data (statistical Mapper)}
\footnotesize On a \alert{point cloud}, a finite set $P \subseteq \R^n$ with Euclidean distance, \alert{clusters} play the role of path components, and the \alert{Mapper graph} is the 1-skeleton of the resulting nerve.
\src{singh2007mapper}
\end{block}
{\small The 1-skeleton keeps vertices and edges but \alert{drops the triangles}. We will come back to this when we count the graph's loops.}
\end{frame}

\subsection{Step 1: the lens}
\begin{frame}{Step 1: PCA, the lens}
Goal: send each pitch shape to $\R^2$, so the cover can be drawn as squares.
\begin{columns}[c]
\begin{column}{0.4\textwidth}\centering
\begin{tikzpicture}[scale=0.85]
\pgfmathsetseed{7}
\begin{scope}[rotate=28]
\foreach \i in {1,...,70}{\pgfmathsetmacro{\a}{2*rand}\pgfmathsetmacro{\b}{0.55*rand}\fill[MiamiRed!60] (\a,\b) circle (1.3pt);}
\draw[-{Latex},very thick] (0,0)--(2.4,0) node[right]{$v_1$};
\draw[-{Latex},very thick] (0,0)--(0,1.0) node[above]{$v_2$};
\end{scope}
\end{tikzpicture}
\par\footnotesize $v_1$: the most variance; $v_2$: the most of what is left
\end{column}
\begin{column}{0.58\textwidth}
\begin{block}{Definition (principal directions)}
\small Center the data: $\mathbf A$ is the $m \times n$ data matrix minus its column means. The first \alert{principal direction} $v_1$ is the unit vector that maximizes $\mathrm{Var}(\mathbf A v)$; each next $v_k$ does the same among unit vectors orthogonal to the earlier ones.
\src{jolliffecadima2016,pearson1901}
\end{block}
\small Centering does not move the $v_k$ (variance ignores shifts). It gives the covariance matrix $S = \mathbf A^{\mathsf T}\mathbf A/(m-1)$, with $\mathrm{Var}(\mathbf A v) = v^{\mathsf T} S v$.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Computing PCA: the SVD}
\begin{block}{Theorem (SVD, matrix version)}
Let $\mathbf A$ have rank $r \ge 1$. Then $\mathbf A = \mathbf U \boldsymbol\Sigma \mathbf V^{\mathsf T}$, where $\mathbf U \in \R^{m \times r}$ and $\mathbf V \in \R^{n \times r}$ have orthonormal columns and $\boldsymbol\Sigma = \mathrm{diag}(\sigma_1 \ge \cdots \ge \sigma_r > 0)$.
\src{axler2024ladr}
\end{block}
\begin{itemize}[<+->]
  \item The columns $v_i$ of $\mathbf V$ are the principal directions, with variance $\lambda_i = \sigma_i^2/(m-1)$. Direction $j$ explains the share $\sigma_j^2 / \sum_l \sigma_l^2$ of the total. \hfill\parencite{jolliffecadima2016}
  \item Our lens: $f(x) = [\,v_1\ v_2\,]^{\mathsf T}(x - \bar x)$.
  \item Why 2: with 10 intervals per axis, 2 gives 100 squares and 3 would give 1{,}000, most empty. And we can draw it.
  \item PCA is \alert{linear}, but clustering uses all 9 coordinates, and 3 other lenses (incl.\ nonlinear Isomap) check it.
\end{itemize}
\end{frame}

\subsection{Step 2: the cover}
\begin{frame}{Step 2: building a cover}
\begin{itemize}
  \item Cover the image $f(P) \subseteq \R^k$ by overlapping hypercubes: equal intervals per axis, neighbors sharing a fixed \% of their length. \hfill\parencite{singh2007mapper,madukpe2026review}
  \item Two parameters control the resolution: the \alert{number of intervals} and the \alert{percentage overlap}. \hfill\parencite{singh2007mapper}
  \item The resulting graph depends on these choices, and tuning them on real data is not easy. \hfill\parencite{carrieremicheloudot2018}
  \item Toy circle: 30\% overlap is under 50\%, so no height lies in three intervals, no three clusters share a point, and the nerve is the graph. On our $10 \times 10$ cover a point can lie in up to four squares, so triangles appear (part 3).
\end{itemize}

\subsection{Step 3: clustering with DBSCAN}
\begin{frame}{Step 3: clustering each preimage}
\begin{itemize}
  \item For each cube $U_i$, cluster the points of $f^{-1}(U_i)$ in the full space $\R^n$. Clusters stand in for path components. \hfill\parencite{singh2007mapper}
  \item We use \alert{DBSCAN}, the second most common choice in reviewed Mapper applications (after hierarchical clustering). \hfill\parencite{madukpe2026review}
  \item Why DBSCAN: it needs no preset number of clusters, finds clusters of any shape, and labels isolated points as \alert{noise}. \hfill\parencite{ester1996dbscan}
\end{itemize}
\vspace{6pt}
\textbf{Setting for the next slides:} a point cloud $P$ with distance $d$, points $p, q \in P$, a radius $\varepsilon > 0$, and a count $M \in \N$ (Eps and MinPts in the original paper).
\end{frame}

\begin{frame}{Density reachability}
\begin{block}{Definition 1 ($\varepsilon$-neighborhood)}
For $p \in P$, $N_\varepsilon(p) = \{\, q \in P : d(p,q) \le \varepsilon \,\}$. \quad $p$ is a \alert{core point} if $|N_\varepsilon(p)| \ge M$ (counting $p$).
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 2 (directly density-reachable)}
For $p, q \in P$: $p$ is directly density-reachable from $q$ if $p \in N_\varepsilon(q)$ and $q$ is a core point.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 3 (density-reachable)}
$p$ is density-reachable from $q$ if there is a chain $q = p_1, p_2, \dots, p_\ell = p$, $\ell \ge 2$, with each $p_{i+1}$ directly density-reachable from $p_i$.
\src{ester1996dbscan}
\end{block}
\note[item]{Why $\ell \ge 2$: with $\ell = 1$ every point is density-reachable from itself, so a lone non-core point is its own cluster and there is no noise. We write $\ell$ because $k$ is the lens dimension.}
\end{frame}

\begin{frame}{Clusters and noise}
\begin{block}{Definition 4 (density-connected)}
$p, q \in P$ are density-connected if both are density-reachable from the same point of $P$.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 5 (cluster)}
A \alert{cluster} is a nonempty $C \subseteq P$ such that
(i) if $p \in C$ and $q$ is density-reachable from $p$, then $q \in C$ (\emph{maximality}),
(ii) any two points of $C$ are density-connected (\emph{connectivity}), and
(iii) $C$ contains a core point.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 6 (noise)}
\alert{Noise}: the points of $P$ in no cluster; it depends on $(\varepsilon, M)$. Ester et al.\ allow a pair per cluster; we, like the algorithm, use one.
\end{block}
\note[item]{Why (iii): without it, a lone border point $b$ next to a core point $c$ passes (i) vacuously (nothing is reachable from a non-core point) and (ii) through $c$, so $\{b\}$ would be a cluster. Ester et al.\ note that every cluster contains a core point; (iii) makes it explicit. A worked example (core, border, noise) and the pseudocode are in backup if asked.}
\end{frame}

\subsection{Step 4: the graph}
\begin{frame}{Step 4: building the graph}
\begin{block}{Construction}
Each cluster is a vertex. Two clusters are joined by an edge whenever they share at least one point, which can happen because neighboring cubes overlap.
\src{singh2007mapper}
\end{block}
\begin{itemize}
  \item The clusters form a cover of (the non-noise part of) $P$, and the graph is the \alert{1-skeleton of its nerve}.
  \item Mapper software such as KeplerMapper outputs this graph only. Three clusters with a common point would form a triangle in the nerve, but the graph keeps just the three edges.
\end{itemize}
\end{frame}

\subsection{Checking a Mapper picture}
\begin{frame}{The nerve's loops are bounded, the graph's are not}
\begin{block}{Theorem (Dey--M\'emoli--Wang)}
If $\mathcal U$ is a finite open cover of a compact, path-connected space $X$ by path-connected sets, the nerve map $\varphi : X \to |N(\mathcal U)|$ induces a surjection $H_1(X) \to H_1(|N(\mathcal U)|)$ ($\Z_2$ coefficients).
For a well-behaved $f$ (finitely many open path components per preimage), Mapper's pullback cover is such a cover, so
\[ b_1(\text{Mapper}) \le b_1(X). \]
\src{deymemoliwang2017}
\end{block}
\begin{itemize}\small
  \item The bound is about the nerve. The graph is only its 1-skeleton, so it can gain loops.
  \item Three disks with a common point (slide on nerves): the disk has $b_1 = 0$, the nerve is a filled triangle ($b_1 = 0$), but the graph is a hollow triangle ($b_1 = 1$).
  \item We test this on our data at the end, when we count the graph's loops.
\end{itemize}
\note[item]{The nerve map: $\varphi(x) = \sum_\alpha \rho_\alpha(x)\, u_\alpha$, with $\{\rho_\alpha\}$ a partition of unity subordinate to $\mathcal U$: each point goes to a weighted average of the vertices of the sets that contain it. $H_1$ of a space is singular homology, which agrees with simplicial homology on $|K|$ (Hatcher Thm.\ 2.27).}
\end{frame}

\begin{frame}{Checking a Mapper picture: persistent loops}
\centering
\includegraphics[width=0.68\textwidth]{figures/rips_toy.pdf}
\raggedright
\begin{block}{Definitions}
\small
\begin{itemize}
  \item \alert{Vietoris--Rips complex} of a finite metric space $(P, \delta)$: $\mathrm{Rips}_t(P) = \{\, \emptyset \ne S \subseteq P : \diam S \le t \,\}$, nested as $t$ grows.
  \item A class is \alert{born} at scale $b$ and \alert{dies} at scale $d$ (filled in, or merged into an older class). \alert{Persistence} $= d - b$. In $H_0$, the classes are components.
  \item \alert{Persistent loop}: an $H_1$ point far from the diagonal of the \alert{persistence diagram}; points near it are usually noise.
\end{itemize}

\end{block}

\begin{frame}{Backup: Reeb graphs}
\note[item]{A multigraph is not a simplicial complex as defined in part 1; subdividing its edges makes it a simple graph, hence a complex.}
\begin{block}{Definition (Reeb graph)}
Let $f : X \to \R$ be continuous. Write $x \sim y$ if $f(x) = f(y)$ and $x, y$ lie in the same connected component of the level set $f^{-1}(f(x))$. The \alert{Reeb graph} is the quotient space $R_f(X) = X/\!\sim$.
\src{carriereoudot2018,reeb1946}
\end{block}
If $X$ is a compact manifold and $f$ is \alert{Morse} (smooth, with invertible Hessian at every critical point \parencite[Def.~1.21]{debray2018morse}), or more generally of \emph{Morse type} (backup), then $R_f(X)$ is a graph, technically a multigraph \parencite[p.~1334]{carriereoudot2018}.

\begin{frame}{Backup: functions of Morse type}
\begin{block}{Definition}
A continuous $f : X \to \R$ on a compact space $X$ is of \alert{Morse type} if
\begin{enumerate}
  \item it has finitely many critical values $a_1 < \dots < a_n$, and over each open interval between them (with $a_0 = -\infty$, $a_{n+1} = +\infty$) there are a compact, locally connected space $Y_i$ and a homeomorphism $\mu_i : Y_i \times (a_i, a_{i+1}) \to f^{-1}(a_i, a_{i+1})$ with $f \circ \mu_i$ the projection onto the second factor;
  \item each $\mu_i$ extends continuously to the closed interval;
  \item every level set has finitely generated homology.
\end{enumerate}
\par\vspace{2pt}\hfill{\scriptsize\textcolor{gray}{\parencite[Def.~1]{carriereoudot2018}}}
\end{block}
Morse functions are of Morse type, but not conversely: a Morse-type function need not be differentiable, and $X$ need not be a manifold. \hfill\parencite{carriereoudot2018}
\end{frame}

\begin{frame}{Backup: the DBSCAN algorithm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{algorithmic}[1]
\small
\State all points start \textsc{unclassified}
\For{each point $p$}
  \If{$p$ is \textsc{unclassified}}
    \If{\Call{ExpandCluster}{$p$}}
      \State start a new cluster id
    \EndIf
  \EndIf
\EndFor
\end{algorithmic}
\vspace{2pt}
{\scriptsize\textcolor{gray}{Main loop adapted from \textcite[p.~229]{ester1996dbscan}.}}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{ExpandCluster($p$):}
\begin{itemize}\small
  \item If $p$ is not a core point, mark it noise for now and return false.
  \item Otherwise give every unclassified or noise point of $N_\varepsilon(p)$ the current cluster id, and repeat from each new core point found.
  \item A border point first marked noise is relabeled if a later cluster reaches it.
\end{itemize}
\end{column}
\end{columns}
\note[item]{Walk through the previous slide's example with the pseudocode: start at a core point, expand, reach the border point, leave the crosses as noise.}
\end{frame}

\begin{frame}{Backup: why DBSCAN is correct}
\begin{block}{Lemma}
If $C$ is a cluster and $p \in C$ is a core point, then $C = \{\, o \in P : o \text{ is density-reachable from } p \,\}$.
\src{ester1996dbscan}
\end{block}
\begin{itemize}
  \item A cluster is determined by \emph{any one} of its core points. ExpandCluster collects everything density-reachable from $p$, so started at a core point it finds the cluster of $p$.
  \item Started at a non-core point it returns false, and nothing is lost: a point in a cluster is reached from that cluster's core points, and a point in no cluster stays noise.
  \item One exception: a border point in two clusters keeps the id of the one found first. Otherwise the result does not depend on the visit order. \hfill\parencite[pp.~229--230]{ester1996dbscan}
  \item Although the paper's Definition 6 allows different $(\varepsilon_i, M_i)$ for each cluster, the algorithm uses \alert{one global pair}, and so do we. \hfill\parencite{ester1996dbscan}
  \item Ours: $\varepsilon = 1.0$ (in standardized units) and $M = 4$.
\end{itemize}
\end{frame}

\begin{frame}{Backup: topological spaces}
\begin{block}{Definition (topology)}
A \alert{topology} on a set $X$ is a collection $\tau$ of subsets of $X$ (the \emph{open sets}) such that
\begin{enumerate}
  \item $\emptyset \in \tau$ and $X \in \tau$;
  \item any union of sets in $\tau$ is in $\tau$;
  \item any finite intersection of sets in $\tau$ is in $\tau$.
\end{enumerate}
\src{munkres2000topology}
\end{block}
\begin{block}{Definition (subspace topology)}
For $Y \subseteq X$, the \alert{subspace topology} is $\tau_Y = \{\, Y \cap U : U \in \tau \,\}$.
\src{munkres2000topology}
\end{block}
For us, $X$ is always a subset of $\R^n$ with the topology coming from Euclidean distance.
\end{frame}
\begin{frame}{Backup: homotopy}
\begin{block}{Definition (homotopy)}
Continuous maps $f, g : X \to Y$ are \alert{homotopic}, $f \simeq g$, if there is a continuous $F : X \times [0,1] \to Y$ with $F(x,0) = f(x)$ and $F(x,1) = g(x)$ for all $x \in X$.
\src{hatcher2001}
\end{block}
\begin{block}{Definition (homotopy equivalence)}
$X$ and $Y$ are \alert{homotopy equivalent}, $X \simeq Y$, if there are maps $f : X \to Y$, $g : Y \to X$ with $g f \simeq \mathrm{id}_X$ and $f g \simeq \mathrm{id}_Y$.
A space homotopy equivalent to a point is \alert{contractible}.
\src{hatcher2001}
\end{block}
Examples: a disk is contractible; an annulus $\simeq$ a circle; a circle is \emph{not} contractible.
\end{frame}
