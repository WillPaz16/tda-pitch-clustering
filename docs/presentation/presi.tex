\section{Introduction}

\begin{frame}{A pitch shape is a point in a 9-dimensional space}
\begin{itemize}
  \item Every MLB pitch is tracked: velocity, movement, spin, release point, \ldots \hfill\parencite{savantcsvdocs}
  \item Each pitch also gets a \alert{label}: four-seam (FF), sinker (SI), cutter (FC), slider (SL), sweeper (ST), curveball (CU), changeup (CH), \ldots \hfill\parencite{mlbglossary_pitchtypes}
  \item The labels change: in 2023 Statcast added \alert{sweeper} and \alert{slurve} as new classifications \hfill\parencite{mlbglossary_sweeper,mlbglossary_slurve}
  \item A \alert{pitch shape} is one pitcher's average of one pitch type: a point in $\R^9$.
\end{itemize}
\vspace{6pt}
\begin{block}{Question}
Do pitch shapes fall into separate clumps, so each label names its own group? Or do they form one connected space that the labels divide, perhaps with a few dense regions?
\end{block}
\vspace{4pt}
We answer with the \alert{Mapper algorithm}, a tool from topological data analysis, and then we check whether what it shows us is real.
\end{frame}

\begin{frame}{Roadmap}
\begin{enumerate}
  \item \textbf{Topological foundations}: simplicial complexes, graphs as complexes, nerves, homology, Reeb graphs
  \item \textbf{The Mapper algorithm}: lens (PCA), cover, clustering (DBSCAN), graph
  \item \textbf{MLB pitch data}: what Mapper shows, and how we check it is real
\end{enumerate}
\vspace{8pt}
\textbf{Thread to watch:} the \emph{nerve} of a cover appears in part 1 as a theorem, in part 2 as the last step of the algorithm, and in part 3 as the reason the graph's loops are not real.
\end{frame}

\begin{frame}{Simplices}
\begin{block}{Definition ($n$-simplex)}
Points $v_0,\dots,v_n \in \R^m$ are \alert{affinely independent} if $v_1 - v_0, \dots, v_n - v_0$ are linearly independent. Their convex hull
\[
[v_0,\dots,v_n] = \Big\{ \textstyle\sum_{i=0}^n t_i v_i \;:\; t_i \ge 0,\ \sum_i t_i = 1 \Big\}
\]
is an \alert{$n$-simplex} with \emph{vertices} $v_i$. The simplex spanned by any nonempty subset of the vertices is a \alert{face}.
\src{hatcher2001}
\end{block}

\begin{frame}{Simplicial complexes}
\begin{block}{Definition (simplicial complex)}
A \alert{simplicial complex} $K$ is a finite collection of simplices such that
\begin{enumerate}
  \item every face of a simplex of $K$ is in $K$;
  \item any two simplices of $K$ are disjoint or intersect in a common face.
\end{enumerate}
Its \alert{vertex set} is the set of its 0-simplices.
\src{edelsbrunnerletscherzomorodian2002}
\end{block}
\begin{block}{Abstract version}
An \alert{abstract simplicial complex} on a vertex set $V$ is a collection $S$ of finite nonempty subsets of $V$ with $\{v\} \in S$ for all $v$, and $\tau \subseteq \sigma \in S \Rightarrow \tau \in S$. A $k$-simplex is a set of $k+1$ vertices. Placing the vertices in general position and filling in each simplex gives a geometric complex; the union of its simplices, as a space, is the \alert{geometric realization} $|S|$.
\src{zomorodiancarlsson2005}
\end{block}
\end{frame}

\begin{frame}{Graphs are simplicial complexes}
\begin{block}{Definition ($k$-skeleton)}
The \alert{$k$-skeleton} $K^{(k)}$ of a simplicial complex $K$ is the subcomplex of all simplices of dimension $\le k$; the 0-skeleton is the vertex set.
\src{edelsbrunnerharerpatel2008}
\end{block}
By the abstract definition, a finite graph $G = (V, E)$ \emph{is} a simplicial complex of dimension $\le 1$ (vertices = 0-simplices, edges = 1-simplices); its realization $|G|$ is a \alert{topological graph}.\par

\begin{frame}{The nerve of a cover}

Let $\mathcal U = \{U_\alpha\}_{\alpha \in A}$ be a finite cover of a space $X$ by nonempty sets. The \alert{nerve} $N(\mathcal U)$ is the abstract simplicial complex with vertex set $A$, where a set $\{\alpha_0,\dots,\alpha_k\} \subseteq A$ spans a $k$-simplex if and only if
\[ U_{\alpha_0} \cap \cdots \cap U_{\alpha_k} \ne \emptyset. \]
\src{singh2007mapper,hatcher2001}
\end{block}

\begin{frame}{The Nerve Theorem}
\begin{block}{Definitions}
\small $C \subseteq \R^d$ is \alert{convex} if the segment between any two points of $C$ lies in $C$.
Spaces $X \simeq Y$ are \alert{homotopy equivalent} if each can be continuously deformed into the other (backup); \emph{contractible} means $\simeq$ a point. E.g.\ annulus $\simeq$ circle, disk $\simeq$ point.
$|N(\mathcal C)|$ is the \alert{geometric realization} of the nerve: the abstract complex drawn as a space, like the triangles on the previous slide.
\src{hatcher2001}
\end{block}
\begin{block}{Theorem (Nerve Theorem, closed convex version; cited, not proved here)}
If $X \subseteq \R^d$ is the union of finitely many closed convex sets $\mathcal C = \{C_1,\dots,C_n\}$, then $X \simeq |N(\mathcal C)|$.
\par\vspace{2pt}\hfill{\scriptsize\textcolor{gray}{\parencite[Thm.~3.9]{bauerkerberrollrolle2023}; an early closed-cover version: \parencite[Cor.~3]{borsuk1948}}}
\end{block}
\begin{itemize}\footnotesize
  \item Why it works: intersections of convex sets are convex, hence empty or contractible.
  \item Open-cover version: if $\mathcal U$ is an open cover of a paracompact $X$ and every nonempty finite intersection is contractible, then $X \simeq |N(\mathcal U)|$. \hfill\parencite{hatcher2001}
  \item \alert{Warning for later:} covers in data analysis are rarely this nice, so a nerve need not have the shape of the space.
\end{itemize}
\end{frame}

\begin{frame}{Homology: the boundary map}
\begin{block}{Definition ($\Z_2$ homology of a complex $K$)}
$C_k$ is the $\Z_2$-vector space with basis the $k$-simplices of $K$. The \alert{boundary map} $\partial_k : C_k \to C_{k-1}$ is the linear map sending each simplex to the sum of its $(k-1)$-dimensional faces:
\[ \partial_1[a,b] = [a] + [b], \qquad \partial_2[a,b,c] = [b,c] + [a,c] + [a,b]. \]
Then $\partial_k \partial_{k+1} = 0$, the \alert{homology} is $H_k = \ker \partial_k / \operatorname{im} \partial_{k+1}$, and the \alert{Betti number} is $b_k = \dim H_k$.
\src{edelsbrunnerletscherzomorodian2002,hatcher2001}
\end{block}
\begin{itemize}
  \item $b_0$ = number of path components. \hfill\parencite{hatcher2001}
  \item A \alert{loop} is a class in $H_1$: a set of edges with zero boundary (every vertex meets an even number of them), up to boundaries of triangles.
  \item $b_1$ counts independent loops.
\end{itemize}
\end{frame}

\begin{frame}{Counting holes: Betti numbers}
With $V$ vertices, $E$ edges, $C$ components, rank--nullity (with $\partial_0 = 0$) gives $C = b_0 = V - \rank \partial_1$, so
\[ b_1 = (E - \rank \partial_1) - \rank \partial_2 = E - V + C - \rank \partial_2. \]
A graph has no triangles, so $b_1(G) = E - V + C$ (agrees with the Euler characteristic).
\par\vspace{8pt}
{\centering
\begin{tikzpicture}[scale=0.8, v/.style={circle, fill=black, inner sep=1.4pt}, lab/.style={font=\small}]
  \draw[thick] (0,0) node[v]{} node[lab,below left]{$a$} -- (1.6,0) node[v]{} node[lab,below right]{$b$} -- (0.8,1.3) node[v]{} node[lab,above]{$c$} -- cycle;

$z = [a,b] + [b,c] + [a,c]$ has $\partial_1 z = 0$: a loop in $G$. In $K$, $z = \partial_2[a,b,c]$: the triangle \emph{fills} it.
\end{frame}


\begin{frame}{Reeb graphs}
\begin{block}{Definition (Reeb graph)}
Let $f : X \to \R$ be continuous. Write $x \sim y$ if $f(x) = f(y)$ and $x, y$ lie in the same connected component of the level set $f^{-1}(f(x))$. The \alert{Reeb graph} is the quotient space $R_f(X) = X/\!\sim$.
\src{carriereoudot2018,reeb1946}
\end{block}
If $X$ is a compact manifold and $f$ is \alert{Morse} (smooth, with invertible Hessian at every critical point \parencite[Def.~1.21]{debray2018morse}), or more generally of \emph{Morse type} (backup), then $R_f(X)$ is a graph, technically a multigraph \parencite[p.~1334]{carriereoudot2018}.
\begin{columns}[c]
\begin{column}{0.46\textwidth}\centering
\includegraphics[width=0.82\textwidth]{figures/reeb_torus.pdf}
\end{column}
\begin{column}{0.5\textwidth}\small
Standing torus, $f$ = height: Morse, with a minimum, two saddles and a maximum. Each level set collapses to one point per component: one near the top and bottom, two in the middle band. The Reeb graph is a circle with two whiskers.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Mapper: from spaces to data}
\begin{block}{Definition (Mapper)}
\footnotesize Let $f : X \to Z$ be continuous and $\mathcal U = \{U_\alpha\}$ a finite open cover of $Z$. Split each preimage $f^{-1}(U_\alpha)$ into its \alert{path components}. These form the \alert{pullback cover} $f^*\mathcal U$ of $X$, and the \alert{Mapper} is its nerve, $M(\mathcal U, f) = N(f^*\mathcal U)$.
\src{singh2007mapper,deymemoliwang2017}
\end{block}
\begin{block}{On data (statistical Mapper)}
\footnotesize A \alert{point cloud} is a finite set $P \subseteq \R^n$ with Euclidean distance. Clustering plays the role of path components: each preimage $f^{-1}(U_\alpha) \cap P$ is split into \alert{clusters}. The \alert{Mapper graph} is the 1-skeleton of the resulting nerve, with one vertex per cluster and one edge per pair of clusters that share a point.
\src{singh2007mapper}
\end{block}
{\small The 1-skeleton keeps vertices and edges but \alert{drops the triangles}. We will come back to this when we count the graph's loops.}
\note[item]{For $Z = \R$, Mapper is a discrete estimator of the Reeb graph (Carri\`ere, Michel, Oudot 2018).}
\end{frame}

\subsection{Step 1: the lens}
\begin{frame}{Step 1: choosing a lens}
\begin{block}{Lens (filter) function}
A continuous map $f : P \to \R^k$, usually with $k$ small, used to organize the cover.
\src{singh2007mapper}
\end{block}
\begin{itemize}
  \item Among the Mapper applications reviewed by \textcite{madukpe2026review}, \alert{PCA} is the most commonly used lens. We use a 2-dimensional PCA lens.
  \item Other lenses (velocity $\times$ vertical movement, Isomap, a density lens) are used later as a robustness check. \hfill\parencite{madukpe2026review,tenenbaum2000isomap}
\end{itemize}
\end{frame}

\begin{frame}{PCA: principal components and centering}
\begin{block}{Definition (principal components)}
\small Data: $m$ observations of $n$ features, a matrix $\mathbf X \in \R^{m \times n}$. A new variable is a linear combination of the columns, $\mathbf X v$. The \alert{first principal component} is $\mathbf X v_1$, where the unit vector $v_1$ maximizes $\mathrm{Var}(\mathbf X v)$. The \alert{$k$-th} is $\mathbf X v_k$, where $v_k$ maximizes $\mathrm{Var}(\mathbf X v)$ among unit vectors orthogonal to $v_1, \dots, v_{k-1}$; it is uncorrelated with the earlier ones. The $v_k$ are the principal directions.
\src{jolliffecadima2016,pearson1901}
\end{block}
\begin{block}{Why we center}
\small Subtract each column's mean $\bar x_j$ to get the centered matrix $\mathbf A \in \R^{m \times n}$.
\begin{itemize}\small
  \item $\mathbf X v$ and $\mathbf A v$ differ by the constant $\sum_j \bar x_j v_j$, so $\mathrm{Var}(\mathbf X v) = \mathrm{Var}(\mathbf A v)$ for every $v$. So centering does not change the $v_k$. It only shifts each component to mean 0.
  \item What it gains: $S = \mathbf A^{\mathsf T}\mathbf A / (m-1)$ (next slide), so PCA is solved by the SVD of $\mathbf A$.
\end{itemize}
\src{jolliffecadima2016}
\end{block}
\end{frame}

\begin{frame}{The covariance matrix}
\begin{block}{Definition}
$S = \dfrac{1}{m-1}\,\mathbf A^{\mathsf T}\mathbf A \in \R^{n \times n}$: diagonal entries are the feature variances, off-diagonal entries the covariances.
\end{block}
\begin{block}{Fact: $S$ is positive semidefinite}
For any $x \in \R^n$: \quad $x^{\mathsf T} S x = \dfrac{1}{m-1}\, x^{\mathsf T}\mathbf A^{\mathsf T}\mathbf A x = \dfrac{1}{m-1}\,\|\mathbf A x\|^2 \ \ge 0.$
\end{block}
Since $S^{\mathsf T} = S$, the spectral theorem gives real eigenvalues and an orthonormal basis of eigenvectors, and positive semidefiniteness makes the eigenvalues nonnegative. The variance of the data along a unit vector $v$ is $v^{\mathsf T} S v$, maximized by the top eigenvector.
\end{frame}

\begin{frame}{Singular value decomposition}
\begin{block}{Theorem (SVD, matrix version)}
Let $\mathbf A$ have rank $r \ge 1$. Then $\mathbf A = \mathbf U \boldsymbol\Sigma \mathbf V^{\mathsf T}$, where $\mathbf U \in \R^{m \times r}$ and $\mathbf V \in \R^{n \times r}$ have orthonormal columns and $\boldsymbol\Sigma = \mathrm{diag}(\sigma_1 \ge \cdots \ge \sigma_r > 0)$.
\src{axler2024ladr}
\end{block}
\begin{itemize}
  \item The columns of $\mathbf V$ (right singular vectors) are eigenvectors of $\mathbf A^{\mathsf T}\mathbf A$, and the columns of $\mathbf U$ are eigenvectors of $\mathbf A\mathbf A^{\mathsf T}$, both for the nonzero eigenvalues $\sigma_i^2$. \hfill\parencite{jolliffecadima2016}
  \item Hence the eigenvalues of $S$ are $\lambda_i = \sigma_i^2/(m-1)$, and the principal directions are the columns $v_i$ of $\mathbf V$. \hfill\parencite{jolliffecadima2016}
  \item The largest variance is $\lambda_1 = \sigma_1^2/(m-1)$.
\end{itemize}
\end{frame}

\begin{frame}{Projecting to $k$ dimensions}
\begin{itemize}
  \item Keep the first $k$ directions: $\mathbf V_k = [v_1 \cdots v_k] \in \R^{n \times k}$.
  \item The lens is $f(x) = \mathbf V_k^{\mathsf T} (x - \bar x)$, with $\bar x$ the column means, and the score matrix is $\mathbf Z_k = \mathbf A \mathbf V_k = \mathbf U_k \boldsymbol\Sigma_k$.
\end{itemize}
\begin{block}{Best rank-$k$ approximation}
The matrix $\mathbf A_k = \mathbf A\mathbf V_k\mathbf V_k^{\mathsf T} = \mathbf U_k\boldsymbol\Sigma_k\mathbf V_k^{\mathsf T}$ is the closest rank-$\le k$ matrix to $\mathbf A$:
in the operator norm, with error $\sigma_{k+1}$ \parencite{axler2024ladr}, and in the least-squares sense \parencite{jolliffecadima2016}.
\end{block}
\begin{block}{Variance explained}
$\mathrm{PVE}_j = \lambda_j / \sum_l \lambda_l = \sigma_j^2 / \sum_l \sigma_l^2$, \qquad $\mathrm{CVE}_k = \sum_{j \le k}\mathrm{PVE}_j$.
\src{jolliffecadima2016}
\end{block}
\end{frame}

\begin{frame}{Why $k = 2$}
\begin{itemize}
  \item On our data, PC1 explains \alert{38.5\%} of the variance and PC2 \alert{21.0\%} (59.5\% together).
  \item The cover lives in $\R^k$: with 10 intervals per axis, $k=2$ gives 100 cubes, while $k = 3$ would give 1{,}000, most nearly empty.
  \item A 2-D lens can be drawn and read, which matters for interpreting the result.
  \item PCA is a \alert{linear} projection, so curved structure can be flattened. That is acceptable here because clustering still happens in the full 9-dimensional space, and we repeat the stability analysis with three other lenses, including the nonlinear Isomap.
\end{itemize}
\end{frame}

\subsection{Step 2: the cover}
\begin{frame}{Step 2: building a cover}
\begin{itemize}
  \item Cover the image $f(P) \subseteq \R^k$ by overlapping hypercubes: each axis is cut into intervals that overlap their neighbors by a fixed percentage. \hfill\parencite{singh2007mapper,madukpe2026review}
  \item Two parameters control the resolution: the \alert{number of intervals} and the \alert{percentage overlap}. \hfill\parencite{singh2007mapper}
  \item The resulting graph depends on these choices, and tuning them on real data is not easy. \hfill\parencite{carrieremicheloudot2018}
\end{itemize}
\vspace{6pt}

\subsection{Step 3: clustering with DBSCAN}
\begin{frame}{Step 3: clustering each preimage}
\begin{itemize}
  \item For each cube $U_i$, cluster the points of $f^{-1}(U_i)$ in the full space $\R^n$. Clusters stand in for path components. \hfill\parencite{singh2007mapper}
  \item We use \alert{DBSCAN}, the second most common choice in reviewed Mapper applications (after hierarchical clustering). \hfill\parencite{madukpe2026review}
  \item Why DBSCAN: it needs no preset number of clusters, finds clusters of any shape, and labels isolated points as \alert{noise}. \hfill\parencite{ester1996dbscan}
\end{itemize}
\vspace{6pt}
\textbf{Setting for the next slides:} a point cloud $P$ with distance $d$, a radius $\varepsilon > 0$, and a count $M \in \N$ (called Eps and MinPts in the original paper).
\end{frame}

\begin{frame}{Density reachability}
\begin{block}{Definition 1 ($\varepsilon$-neighborhood)}
$N_\varepsilon(p) = \{\, q \in P : d(p,q) \le \varepsilon \,\}$. \quad A point $p$ is a \alert{core point} if $\# N_\varepsilon(p) \ge M$, i.e.\ its neighborhood holds at least $M$ points, counting $p$.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 2 (directly density-reachable)}
$p$ is directly density-reachable from $q$ if $p \in N_\varepsilon(q)$ and $q$ is a core point.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 3 (density-reachable)}
$p$ is density-reachable from $q$ if there is a chain $q = p_1, p_2, \dots, p_k = p$ with each $p_{i+1}$ directly density-reachable from $p_i$.
\src{ester1996dbscan}
\end{block}
\end{frame}

\begin{frame}{Clusters and noise}
\begin{block}{Definition 4 (density-connected)}
$p$ and $q$ are density-connected if both are density-reachable from some common point $o$.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 5 (cluster)}
A \alert{cluster} is a nonempty $C \subseteq P$ such that
(i) if $p \in C$ and $q$ is density-reachable from $p$, then $q \in C$ (\emph{maximality}), and
(ii) any two points of $C$ are density-connected (\emph{connectivity}).
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 6 (noise)}
\alert{Noise} is the set of points of $P$ in no cluster. It depends on the chosen $(\varepsilon, M)$.
\src{ester1996dbscan}
\end{block}
\end{frame}

\begin{frame}{Example: core, border, noise}
\centering
\includegraphics[width=0.72\textwidth]{figures/dbscan_toy.pdf}
\begin{itemize}\small
  \item Solid circle: a \alert{core point}; its $\varepsilon$-ball holds 6 points $\ge M = 4$.
  \item Dashed circle: a \alert{border point}; only 3 points in its ball, but a core point is among them, so it is density-reachable and joins the cluster.
  \item Crosses: \alert{noise}; each is more than $\varepsilon$ from every core point. With $\varepsilon = 0.7$ some would join a cluster: noise depends on $(\varepsilon, M)$.
\end{itemize}
\end{frame}

\begin{frame}{The DBSCAN algorithm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{algorithmic}[1]
\small
\State all points start \textsc{unclassified}
\For{each point $p$}
  \If{$p$ is \textsc{unclassified}}
    \If{\Call{ExpandCluster}{$p$}}
      \State start a new cluster id
    \EndIf
  \EndIf
\EndFor
\end{algorithmic}
\vspace{2pt}
{\scriptsize\textcolor{gray}{Main loop adapted from \textcite[p.~229]{ester1996dbscan}.}}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{ExpandCluster($p$):}
\begin{itemize}\small
  \item If $p$ is not a core point, mark it noise for now and return false.
  \item Otherwise give every unclassified or noise point of $N_\varepsilon(p)$ the current cluster id, and repeat from each new core point found.
  \item A border point first marked noise is relabeled if a later cluster reaches it.
\end{itemize}
\end{column}
\end{columns}
\note[item]{Walk through the previous slide's example with the pseudocode: start at a core point, expand, reach the border point, leave the crosses as noise.}
\end{frame}

\begin{frame}{Why the algorithm is correct}
\begin{block}{Lemma}
If $C$ is a cluster and $p \in C$ is a core point, then $C = \{\, o \in P : o \text{ is density-reachable from } p \,\}$.
\src{ester1996dbscan}
\end{block}
\begin{itemize}
  \item A cluster is determined by \emph{any one} of its core points. ExpandCluster collects everything density-reachable from $p$, so started at a core point it finds the cluster of $p$.
  \item Started at a non-core point it returns false, and nothing is lost: a point in a cluster is reached from that cluster's core points, and a point in no cluster stays noise.
  \item One exception: a border point in two clusters keeps the id of the one found first. Otherwise the result does not depend on the visit order. \hfill\parencite[pp.~229--230]{ester1996dbscan}
  \item Although the paper's Definition 6 allows different $(\varepsilon_i, M_i)$ for each cluster, the algorithm uses \alert{one global pair}, and so do we. \hfill\parencite{ester1996dbscan}
  \item Ours: $\varepsilon = 1.0$ (in standardized units) and $M = 4$.
\end{itemize}
\end{frame}

\subsection{Step 4: the graph}
\begin{frame}{Step 4: building the graph}
\begin{block}{Construction}
Each cluster is a vertex. Two clusters are joined by an edge whenever they share at least one point, which can happen because neighboring cubes overlap.
\src{singh2007mapper}
\end{block}
\begin{itemize}
  \item The clusters form a cover of (the non-noise part of) $P$, and the graph is the \alert{1-skeleton of its nerve}.
  \item Software outputs this graph only. Three clusters with a common point would form a triangle in the nerve, but the graph keeps just the three edges.
\end{itemize}
\end{frame}

\begin{frame}{The graph has 86 loops. Are they real?}
\begin{block}{What ``real'' means}
A loop here is an independent cycle counted by $b_1 = 86$ (not a self-edge). It is \alert{real} if it reflects a hole in the space the pitch shapes sample, and not a by-product of how we built the graph.
\end{block}
\textbf{Why it matters.} A real loop would be a discovery: a ring of pitch shapes around a region of shapes nobody throws, which would need an explanation. Reading an artifact that way would be a false finding.
\par\vspace{8pt}
\textbf{Three tests.} A real loop must
\begin{enumerate}
  \item survive the nerve's triangles;
  \item persist across scales;
  \item beat the null models.
\end{enumerate}
\end{frame}

\begin{frame}{Test 1: do the nerve's triangles fill them?}
\centering
\includegraphics[width=0.56\textwidth]{figures/nerve_b1.pdf}

Graph: $b_1 = E - V + C = 143 - 66 + 9 = 86$. The full nerve adds 113 triangles; $\partial_2$ is a $143 \times 113$ matrix over $\Z_2$ of rank 85, so $b_1 = 86 - 85 = 1$.
\par\vspace{3pt}
Across 27 parameter choices, graph $b_1$ ranges 33--147; nerve $b_1$ stays 0--4. The null models' nerves give 2--9 (Gaussian) and 0--2 (shuffle).
\end{frame}

\begin{frame}{Test 2: what is a persistent loop?}
\centering
\includegraphics[width=0.84\textwidth]{figures/rips_toy.pdf}
\raggedright
\begin{block}{Definition}
\small For a finite metric space $(P, d)$, $\mathrm{Rips}_t(P) = \{\, \emptyset \ne S \subseteq P : \diam S \le t \,\}$; these complexes are nested as $t$ grows \parencite{bauer2021ripser}. A class in $H_1$ is \alert{born} at the scale $b$ where it first appears and \alert{dies} at the scale $d$ where triangles fill it, or it merges with an older class. Its \alert{persistence} is $d - b$. A \alert{persistent loop} is one with large persistence: a point far from the diagonal in the \alert{persistence diagram} of pairs $(b, d)$. Points near the diagonal are usually read as noise. \parencite{edelsbrunnerletscherzomorodian2002,zomorodiancarlsson2005,fasy2014confidence}
\end{block}
\note[item]{The loop in the figure dies at $d = 1.62$. The third panel is drawn at $t = 1.94$ because a Rips complex can fill a loop before its flat picture covers the center.}
\end{frame}

\begin{frame}{Why: nerves can only kill loops}
\begin{block}{Theorem (Dey--M\'emoli--Wang; cited, not proved here)}
If $\mathcal U$ is a finite open cover of a compact space $X$ by path-connected sets, the nerve map induces a surjection $H_1(X) \to H_1(N(\mathcal U))$ ($\Z_2$ coefficients).
For a well-behaved $f$ (finitely many open path components per preimage), Mapper's pullback cover is such a cover, so
\[ b_1(\text{Mapper}) \le b_1(X). \]
\src{deymemoliwang2017}
\end{block}
\begin{itemize}\small
  \item The pitch shapes show \alert{no persistent loop}, just like a Gaussian null. This is consistent with $b_1(X) = 0$, and then every Mapper nerve of $X$ would have $b_1 = 0$.
  \item The 86 graph loops come from reading only the 1-skeleton: triangles fill 85 of them.
  \item The theorem needs a cover of a space by path-connected sets. We have DBSCAN clusters, which are not path components, on a finite sample, so it does not strictly apply, and our nerve keeps $b_1$ of 0--4 (the nulls' nerves give 0--9). We use the theorem as support for the evidence, not as a proof about the data.
\end{itemize}
\end{frame}

\begin{frame}{Three dense regions}
To look for dense regions, we follow \textcite{maloof2026classify}: persistence of the superlevel sets of a smoothed density of \emph{single pitches}, in speed relative to the pitcher's 95th percentile that game, horizontal break, and induced vertical break.
\begin{itemize}
  \item \alert{Three robust peaks}: fastballs, breaking balls, and changeups (relative persistence 0.17 and 0.08 for the smaller two). Each beats all 50 same-covariance Gaussian replicates ($p = 0.02$, the smallest 50 allow).
  \item They survive every smoothing level (1, 1.5, 2 cells) and every random half of the data, close to where \textcite{maloof2026classify} found them for college pitches.
  \item A small curveball-like peak (0.013) is also stable. A sweeper-like bump (0.012) is not, and we do not claim it.
  \item Along PC1 of the pitch shapes, two peaks: a fastball side and a breaking-ball side.
\end{itemize}
\vspace{4pt}
So the space is connected but not featureless: \alert{one continuum with a few dense regions}.
\end{frame}

\begin{frame}{Labels cut a continuum, so grading within a label is fragile}
\begin{itemize}\footnotesize
  \item Each label covers a region of shape space, and neighboring regions touch. Statcast itself relabeled some of our shapes between our fit and a later re-pull (SL$\leftrightarrow$ST, SI$\leftrightarrow$FF).
  \item Our Stuff+-style models (CatBoost, trained on 2022--24) grade a pitch against its own type on a $100 \pm 15$ scale. The 2025 and 2026 seasons are out of sample.
  \item A \alert{boundary pitch} (at least 150 pitches) has at least half of its 20 nearest other-pitcher shapes, in flight space, carrying one other label. We grade it under both labels.
\end{itemize}

\begin{frame}{Conclusions}
\begin{block}{About pitches}
Pitch-shape space is one \alert{connected continuum} apart from the very slowest pitches, which sit apart by absolute speed and delivery, not by flight shape. It has no gaps beyond a Gaussian, no loop that replicates across seasons, about 6 of 9 dimensions, and three dense regions at the pitch level. Labels divide it, and grading within a label is fragile near boundaries.
\end{block}
\begin{block}{About the method}
\begin{itemize}
  \item Mapper gives a picture we can read. We then \alert{check it with null models, homology, and dimension estimates}.
  \item Count loops in the full nerve, because the graph is only its 1-skeleton. For covers by path-connected sets, Dey--M\'emoli--Wang bound the nerve's loops by those of the space.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Future work}
\begin{itemize}
  \item \textbf{Pitch development.} Two exploratory tests designed in advance (backup) suggest that more typical shapes get fewer whiffs than their physics predicts. A planner that routes a pitch through shapes others already throw, scored by a Stuff+-style model \parencite{mcgrattan2023stuffplus}, could be tested prospectively.
  \item \textbf{Arm slot.} Model movement \emph{relative to} a pitcher's arm angle instead of raw movement. \hfill\parencite{savantcsvdocs}
  \item \textbf{Replication.} Repeat on other seasons, and require a minimum number of pitches per shape.
\end{itemize}
\end{frame}

\begin{frame}{Backup: functions of Morse type}
\begin{block}{Definition}
A continuous $f : X \to \R$ on a compact space $X$ is of \alert{Morse type} if
\begin{enumerate}
  \item it has finitely many critical values $a_1 < \dots < a_n$, and over each open interval between them (with $a_0 = -\infty$, $a_{n+1} = +\infty$) there are a compact, locally connected space $Y_i$ and a homeomorphism $\mu_i : Y_i \times (a_i, a_{i+1}) \to f^{-1}(a_i, a_{i+1})$ with $f \circ \mu_i$ the projection onto the second factor;
  \item each $\mu_i$ extends continuously to the closed interval;
  \item every level set has finitely generated homology.
\end{enumerate}
\par\vspace{2pt}\hfill{\scriptsize\textcolor{gray}{\parencite[Def.~1]{carriereoudot2018}}}
\end{block}
Morse functions are of Morse type, but not conversely: a Morse-type function need not be differentiable, and $X$ need not be a manifold. \hfill\parencite{carriereoudot2018}
\end{frame}

\begin{frame}{Backup: topological spaces}
\begin{block}{Definition (topology)}
A \alert{topology} on a set $X$ is a collection $\tau$ of subsets of $X$ (the \emph{open sets}) such that
\begin{enumerate}
  \item $\emptyset \in \tau$ and $X \in \tau$;
  \item any union of sets in $\tau$ is in $\tau$;
  \item any finite intersection of sets in $\tau$ is in $\tau$.
\end{enumerate}
\src{munkres2000topology}
\end{block}
\begin{block}{Definition (subspace topology)}
For $Y \subseteq X$, the \alert{subspace topology} is $\tau_Y = \{\, Y \cap U : U \in \tau \,\}$.
\src{munkres2000topology}
\end{block}
For us, $X$ is always a subset of $\R^n$ with the topology coming from Euclidean distance.
\end{frame}
\begin{frame}{Backup: homotopy}
\begin{block}{Definition (homotopy)}
Continuous maps $f, g : X \to Y$ are \alert{homotopic}, $f \simeq g$, if there is a continuous $F : X \times [0,1] \to Y$ with $F(x,0) = f(x)$ and $F(x,1) = g(x)$ for all $x \in X$.
\src{hatcher2001}
\end{block}
\begin{block}{Definition (homotopy equivalence)}
$X$ and $Y$ are \alert{homotopy equivalent}, $X \simeq Y$, if there are maps $f : X \to Y$, $g : Y \to X$ with $g f \simeq \mathrm{id}_X$ and $f g \simeq \mathrm{id}_Y$.
A space homotopy equivalent to a point is \alert{contractible}.
\src{hatcher2001}
\end{block}
Examples: a disk is contractible; an annulus $\simeq$ a circle; a circle is \emph{not} contractible.
\end{frame}
