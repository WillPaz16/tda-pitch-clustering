\begin{frame}{Simplices}
\note[item]{\textbf{Say:} A simplex is the simplest shape in each dimension: a point, a segment, a triangle, a tetrahedron. Affinely independent just means the points don't collapse into a lower dimension, so three points not on a line. The simplex is every weighted average of its vertices, and a face is the simplex on any subset of them.}
\begin{block}{Definition ($n$-simplex)}
Points $v_0,\dots,v_n \in \R^m$ are \alert{affinely independent} if $v_1 - v_0, \dots, v_n - v_0$ are linearly independent. If they are, their convex hull
\[
[v_0,\dots,v_n] = \Big\{ \textstyle\sum_{i=0}^n t_i v_i \;:\; t_i \ge 0,\ \sum_i t_i = 1 \Big\}
\]
is an \alert{$n$-simplex} with \emph{vertices} $v_i$. The simplex spanned by any nonempty subset of the vertices is a \alert{face}.
\src{hatcher2001}
\end{block}

\begin{frame}{Simplicial complexes}
\note[item]{\textbf{Say:} A simplicial complex glues simplices together nicely: every face is included, and two simplices only meet along a shared face. The abstract version forgets the geometry and only remembers which sets of vertices form simplices. Example: three edges make a hollow triangle; add the set $\{a,b,c\}$ and it is filled. Drawing an abstract complex gives back a geometric one, its realization.}
\begin{block}{Definition (simplicial complex)}
A \alert{simplicial complex} $K$ is a finite collection of simplices such that (1) every face of a simplex of $K$ is in $K$, and (2) any two simplices of $K$ are disjoint or intersect in a common face. Its \alert{vertex set} is the set of its 0-simplices.
\src{edelsbrunnerletscherzomorodian2002}
\end{block}
\begin{block}{Definition (abstract simplicial complex)}
An \alert{abstract simplicial complex} $K$ on a finite vertex set $V$ is a collection of nonempty subsets of $V$, its simplices, such that
(1) $\{v\} \in K$ for every $v \in V$, and (2) every nonempty subset of a simplex in $K$ is in $K$. A simplex with $k+1$ vertices is a $k$-simplex.
\src{zomorodiancarlsson2005}
\end{block}
{\small Example: $\{\{a\},\{b\},\{c\},\{a,b\},\{b,c\},\{a,c\}\}$ is a hollow triangle; adding $\{a,b,c\}$ fills it. Drawing $K$ gives its \alert{geometric realization} $|K|$.}
\end{frame}

\begin{frame}{Graphs are simplicial complexes}
\note[item]{\textbf{Say:} A simple graph is just an abstract complex with nothing above dimension 1: vertices and edges. The $k$-skeleton keeps the simplices of dimension at most $k$. The key point for later: $G$ has a loop $abca$, and in $K$ the triangle fills it. So a graph, a 1-skeleton, can show a loop the full complex does not have. That is part 3 in miniature.}
\note[item]{Simple: no edge from a vertex to itself and no repeated edges. An abstract complex cannot hold either, so a multigraph must be subdivided first.}
\begin{block}{Definition ($k$-skeleton)}
The \alert{$k$-skeleton} $K^{(k)}$ of a simplicial complex $K$ is the subcomplex of all simplices of dimension $\le k$; the 0-skeleton is the vertex set.
\src{edelsbrunnerharerpatel2008}
\end{block}
By the abstract definition, a finite \alert{simple} graph $G = (V, E)$ \emph{is} a simplicial complex of dimension $\le 1$; its realization $|G|$ is a \alert{topological graph}.\par

\begin{frame}{The nerve of a cover}
\note[item]{\textbf{Say:} Given a cover, the nerve has one vertex per set, and a simplex whenever those sets share a common point. Left: three arcs cover a circle; each pair overlaps but no point is in all three, so the nerve is a hollow triangle, which has the shape of a circle. Right: three disks with a common point give a filled triangle, which shrinks to a point, like the disk.}
\begin{block}{Definition (nerve)}
Let $\mathcal U = \{U_\alpha\}_{\alpha \in A}$ be a finite cover of a space $X$ by nonempty sets. The \alert{nerve} $N(\mathcal U)$ is the abstract simplicial complex with vertex set $A$, where a set $\{\alpha_0,\dots,\alpha_k\} \subseteq A$ spans a $k$-simplex if and only if
\[ U_{\alpha_0} \cap \cdots \cap U_{\alpha_k} \ne \emptyset. \]
\src{singh2007mapper,hatcher2001}
\end{block}

\begin{frame}{The Nerve Theorem}
\note[item]{\textbf{Say:} Homotopy equivalent means one space can be deformed into the other: maps both ways whose composites can be continuously deformed back to the identity. Contractible means it can be shrunk to a point. The theorem: if every intersection in the cover is contractible, the nerve has the same shape as the space. The warning: covers built from data are rarely this nice, so later the nerve need not match the space.}
\begin{block}{Definition (homotopy equivalence)}
Spaces $X$ and $Y$ are \alert{homotopy equivalent}, $X \simeq Y$, if there are continuous maps $f : X \to Y$ and $g : Y \to X$ such that $g \circ f$ and $f \circ g$ are homotopic to the identity maps. A space is \alert{contractible} if it is homotopy equivalent to a point.
\src{hatcher2001}
\end{block}
\begin{block}{Theorem (Nerve Theorem)}
If $X \subseteq \R^n$ and $\mathcal U$ is an open cover of $X$ in which every nonempty finite intersection of sets is contractible, then $X \simeq |N(\mathcal U)|$, the geometric realization of the nerve.
\par\vspace{2pt}\hfill{\scriptsize\textcolor{gray}{\parencite[Cor.~4G.3]{hatcher2001}, stated for paracompact spaces, which include every subset of $\R^n$ \parencite[Thm.~41.4]{munkres2000topology}}}
\end{block}
\begin{itemize}\footnotesize
  \item \alert{Warning for later:} covers in data are rarely this nice, so a nerve need not match the space.
\end{itemize}
\note[item]{The circle's arcs on the nerve slide are open, so the theorem covers that example. Closed convex covers also work, since convex intersections are contractible \parencite[Thm.~3.9]{bauerkerberrollrolle2023}.}
\end{frame}

\begin{frame}{Homology and Betti numbers}
\note[item]{\textbf{Say:} Homology counts holes with linear algebra. $C_k$ is formal sums of $k$-simplices over $\Z_2$, so each simplex is simply in or out. The boundary of a simplex is the sum of its faces: the boundary of triangle $abc$ is $bc + ac + ab$. The boundary of a boundary is zero. $H_k$ is cycles modulo boundaries: loops that are not filled in. $b_0$ counts components and $b_1$ counts independent loops; for a graph, $b_1 = E - V + C$. In the picture, $G$ has one loop, and $K$ fills it with the triangle, so $b_1 = 0$.}
\begin{block}{Definition ($\Z_2$ homology of a complex $K$)}
$C_k$ is the $\Z_2$-vector space with basis the $k$-simplices of $K$. The \alert{boundary map} $\partial_k : C_k \to C_{k-1}$ sends each simplex to the sum of its faces, e.g.\ $\partial_2\{a,b,c\} = \{b,c\} + \{a,c\} + \{a,b\}$. Since $\partial_k \partial_{k+1} = 0$, define the \alert{$k$-th homology group}
\[ H_k = \ker \partial_k / \operatorname{im} \partial_{k+1}, \qquad \text{and the \alert{$k$-th Betti number} } b_k = \dim H_k. \]
\src{edelsbrunnerletscherzomorodian2002,hatcher2001}
\end{block}
\begin{itemize}\small
  \item $b_0$ counts components and $b_1$ independent \alert{loops} (for a graph, its \alert{cycle rank}; never self-edges).
  \item Rank--nullity: $b_1 = E - V + C - \rank \partial_2$; a graph has no triangles, so $b_1(G) = E - V + C$.
\end{itemize}

  \draw[thick] (0,0) node[v]{} node[lab,below left]{$a$} -- (1.6,0) node[v]{} node[lab,below right]{$b$} -- (0.8,1.3) node[v]{} node[lab,above]{$c$} -- cycle;

  \begin{scope}[xshift=7.2cm]

  \end{scope}

\note[item]{Point at the pictures: $z = \{a,b\} + \{b,c\} + \{a,c\}$ has $\partial_1 z = 0$, a loop in $G$. In $K$, $z = \partial_2\{a,b,c\}$, so the triangle fills it.}
\note[item]{Rank--nullity: with $\partial_0 = 0$, $C = b_0 = V - \rank \partial_1$, and $b_1 = (E - \rank \partial_1) - \rank \partial_2$. For a graph this agrees with the Euler characteristic.}
\end{frame}


\begin{frame}{Mapper: from spaces to data}
\note[item]{\textbf{Say:} Mapper: take a map $f$ from $X$ to a simpler space, cover that space, pull each set back to $X$, and split it into path components. Those pieces cover $X$, and Mapper is their nerve. In the picture, $f$ is height on a circle with three intervals; the middle one pulls back to two arcs, so the nerve is a loop, like the circle. On data, clusters stand in for path components, and software keeps only the graph, the 1-skeleton. Remember that dropping triangles is where extra loops come from.}
\begin{columns}[c]
\begin{column}{0.56\textwidth}
\begin{block}{Definition (Mapper)}
\small Let $f : X \to Z$ be continuous and $\mathcal U = \{U_\alpha\}$ a finite open cover of $Z$. Split each preimage $f^{-1}(U_\alpha)$ into its \alert{path components}. These form the \alert{pullback cover} $f^*\mathcal U$ of $X$, and the \alert{Mapper} is its nerve, $M(\mathcal U, f) = N(f^*\mathcal U)$.
\src{singh2007mapper,deymemoliwang2017}
\end{block}
\begin{block}{On data (statistical Mapper)}
\small On a \alert{point cloud}, a finite set $P \subseteq \R^n$ with Euclidean distance, \alert{clusters} play the role of path components, and the \alert{Mapper graph} is the 1-skeleton of the nerve.
\src{singh2007mapper}
\end{block}
\end{column}

\begin{frame}{Statistical Mapper: overview}
\note[item]{\textbf{Say:} Four steps: lens, cover, cluster, graph. The lens maps the data to a low-dimensional space; ours is PCA. We cover the lens image with overlapping boxes, cluster each preimage using all the coordinates, not just the lens, and connect clusters that share points. The lens only decides which points get clustered together.}

  \item Steps 1--3 build a cover of $P$ (apart from noise points), and step 4 keeps the 1-skeleton of its nerve. \hfill\parencite{singh2007mapper}
  \item The \alert{lens} is a continuous map $f : P \to \R^k$, $k$ small, that organizes the cover; PCA is the most common choice and ours. \hfill\parencite{madukpe2026review}

\subsection{Step 1: the lens}
\begin{frame}{Step 1: PCA, the lens}
\note[item]{\textbf{Say:} We need a lens to $\R^2$ so the cover is made of squares. PCA finds the direction of most variance, $v_1$, then the direction of most remaining variance orthogonal to it, $v_2$. Centering does not change the directions, but the SVD maximizes $\|\mathbf A v\|^2$, and only after centering is that the variance; uncentered, $v_1$ would point at the average pitch.}
Goal: send each pitch shape to $\R^2$, so the cover can be drawn as squares.
\begin{columns}[c]
\begin{column}{0.4\textwidth}\centering
\begin{tikzpicture}[scale=0.85]
\pgfmathsetseed{7}
\begin{scope}[rotate=28]
\foreach \i in {1,...,70}{\pgfmathsetmacro{\a}{2*rand}\pgfmathsetmacro{\b}{0.55*rand}\fill[MiamiRed!60] (\a,\b) circle (1.3pt);}
\draw[-{Latex},very thick] (0,0)--(2.4,0) node[right]{$v_1$};
\draw[-{Latex},very thick] (0,0)--(0,1.0) node[above]{$v_2$};
\end{scope}
\end{tikzpicture}
\par\footnotesize $v_1$: the most variance; $v_2$: the most of what is left
\end{column}
\begin{column}{0.58\textwidth}
\begin{block}{Definition (principal directions)}
\small Center the data: $\mathbf A$ is the $m \times n$ data matrix minus its column means. The first \alert{principal direction} $v_1$ is the unit vector that maximizes $\mathrm{Var}(\mathbf A v)$; each next $v_k$ does the same among unit vectors orthogonal to the earlier ones.
\src{jolliffecadima2016,pearson1901}
\end{block}
\small Centering does not move the $v_k$, but it makes $\|\mathbf A v\|^2 = (m-1)\,\mathrm{Var}(\mathbf A v) = (m-1)\,v^{\mathsf T} S v$, with $S = \mathbf A^{\mathsf T}\mathbf A/(m-1)$ the covariance matrix. \alert{So we want the unit $v$ that maximizes $\|\mathbf A v\|^2$.}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Computing PCA: the SVD}
\note[item]{\textbf{Say:} To compute PCA we use the SVD. Squaring it gives $S$ in an orthonormal basis, so $v^{\mathsf T} S v$ is a weighted average of the $\lambda_i$, largest at $v_1$; orthogonal to $v_1$, largest at $v_2$. So the columns of $V$ are the principal directions, and the squared singular values give each one's share of variance. Our lens projects each centered pitch shape onto $v_1$ and $v_2$. Why two: three would give a thousand boxes, mostly empty, and we can draw two. PCA is linear, but clustering uses all nine coordinates, and three other lenses, including nonlinear Isomap, check it.}
\begin{block}{Theorem (SVD, matrix version)}
Let $\mathbf A$ have rank $r \ge 1$. Then $\mathbf A = \mathbf U \boldsymbol\Sigma \mathbf V^{\mathsf T}$, where $\mathbf U \in \R^{m \times r}$ and $\mathbf V \in \R^{n \times r}$ have orthonormal columns and $\boldsymbol\Sigma = \mathrm{diag}(\sigma_1 \ge \cdots \ge \sigma_r > 0)$.
\src{axler2024ladr}
\end{block}
\begin{itemize}[<+->]
  \item $\mathbf A^{\mathsf T}\mathbf A = \mathbf V\boldsymbol\Sigma^2\mathbf V^{\mathsf T}$, so $v^{\mathsf T} S v = \sum_i \lambda_i \langle v, v_i\rangle^2$ with $\lambda_i = \sigma_i^2/(m-1)$. This is at most $\lambda_1$, with equality at $v_1$: \alert{the columns of $\mathbf V$ are the principal directions.}
  \item $v_j$ explains $\sigma_j^2 / \sum_l \sigma_l^2$ of the variance (ours: 38.5\%, 21.0\%). \hfill\parencite{jolliffecadima2016}
  \item Our lens: $f(x) = [\,v_1\ v_2\,]^{\mathsf T}(x - \bar x)$.
  \item Why 2: 100 squares; 3 would give 1{,}000, most empty. And we can draw it.
  \item PCA is \alert{linear}, but clustering uses all 9 coordinates, and 3 other lenses (incl.\ nonlinear Isomap) check it.
\end{itemize}
\end{frame}

\subsection{Step 2: the cover}
\begin{frame}{Step 2: building a cover}
\note[item]{\textbf{Say:} Cover the lens image with overlapping squares. The number of intervals and the percent overlap set the resolution, and the graph depends on them, so we check many settings later. In one dimension with under 50\% overlap, no three intervals meet, so there are no triangles. In our 10 by 10 grid up to four squares can overlap, so triangles do appear. That matters in part 3.}
\begin{itemize}
  \item Cover the image $f(P) \subseteq \R^k$ by overlapping hypercubes: equal intervals per axis, neighbors sharing a fixed \% of their length. \hfill\parencite{singh2007mapper,madukpe2026review}
  \item Two parameters control the resolution: the \alert{number of intervals} and the \alert{percentage overlap}. \hfill\parencite{singh2007mapper}
  \item The resulting graph depends on these choices, and tuning them on real data is not easy. \hfill\parencite{carrieremicheloudot2018}
  \item Toy circle: 30\% overlap is under 50\%, so no height lies in three intervals, no three clusters share a point, and the nerve is the graph. On our $10 \times 10$ cover a point can lie in up to four squares, so triangles appear (part 3).
\end{itemize}

\subsection{Step 3: clustering with DBSCAN}
\begin{frame}{Step 3: clustering each preimage}
\note[item]{\textbf{Say:} In each square, cluster the points that land there, in the full 9-dimensional space. Clusters play the role of path components. We use DBSCAN: no preset number of clusters, any shape, and it labels isolated points as noise. It has two parameters, a radius $\varepsilon$ and a count $M$.}
\begin{itemize}
  \item For each cube $U_i$, cluster the points of $f^{-1}(U_i)$ in the full space $\R^n$. Clusters stand in for path components. \hfill\parencite{singh2007mapper}
  \item We use \alert{DBSCAN}, the second most common choice in reviewed Mapper applications (after hierarchical clustering). \hfill\parencite{madukpe2026review}
  \item Why DBSCAN: it needs no preset number of clusters, finds clusters of any shape, and labels isolated points as \alert{noise}. \hfill\parencite{ester1996dbscan}
\end{itemize}
\vspace{6pt}
\textbf{Setting for the next slides:} a point cloud $P$ with distance $d$, points $p, q \in P$, a radius $\varepsilon > 0$, and a count $M \in \N$ (Eps and MinPts in the original paper).
\end{frame}

\begin{frame}{Density reachability}
\note[item]{\textbf{Say:} DBSCAN in plain words. A core point has at least $M$ points within $\varepsilon$ of it, counting itself. From a core point you can step to anything within $\varepsilon$. Density-reachable means there is a chain of such steps. Only core points can be stepped from, which is why the chain needs at least two points and why lone points end up as noise.}
\begin{block}{Definition 1 ($\varepsilon$-neighborhood)}
For $p \in P$, $N_\varepsilon(p) = \{\, q \in P : d(p,q) \le \varepsilon \,\}$. \quad $p$ is a \alert{core point} if $|N_\varepsilon(p)| \ge M$ (counting $p$).
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 2 (directly density-reachable)}
For $p, q \in P$: $p$ is directly density-reachable from $q$ if $p \in N_\varepsilon(q)$ and $q$ is a core point.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 3 (density-reachable)}
$p$ is density-reachable from $q$ if there is a chain $q = p_1, p_2, \dots, p_\ell = p$, $\ell \ge 2$, with each $p_{i+1}$ directly density-reachable from $p_i$.
\src{ester1996dbscan}
\end{block}
\note[item]{Why $\ell \ge 2$: with $\ell = 1$ every point is density-reachable from itself, so a lone non-core point is its own cluster and there is no noise. We write $\ell$ because $k$ is the lens dimension.}
\end{frame}

\begin{frame}{Clusters and noise}
\note[item]{\textbf{Say:} Two points are density-connected if both can be reached from some common point. A cluster is a maximal, connected set that contains a core point. Noise is everything left over, and what counts as noise depends on $\varepsilon$ and $M$. Picture it as dense blobs plus a few stragglers.}
\begin{block}{Definition 4 (density-connected)}
$p, q \in P$ are density-connected if both are density-reachable from the same point of $P$.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 5 (cluster)}
A \alert{cluster} is a nonempty $C \subseteq P$ such that
(i) if $p \in C$ and $q$ is density-reachable from $p$, then $q \in C$ (\emph{maximality}),
(ii) any two points of $C$ are density-connected (\emph{connectivity}), and
(iii) $C$ contains a core point.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 6 (noise)}
\alert{Noise}: the points of $P$ in no cluster; it depends on $(\varepsilon, M)$. Ester et al.\ allow a pair per cluster; we, like the algorithm, use one.
\end{block}
\note[item]{Why (iii): without it, a lone border point $b$ next to a core point $c$ passes (i) vacuously (nothing is reachable from a non-core point) and (ii) through $c$, so $\{b\}$ would be a cluster. Ester et al.\ note that every cluster contains a core point; (iii) makes it explicit. A worked example (core, border, noise) and the pseudocode are in backup if asked.}
\end{frame}

\subsection{Step 4: the graph}
\begin{frame}{Step 4: building the graph}
\note[item]{\textbf{Say:} Each cluster becomes a vertex, and two clusters that share a point get an edge. Because neighboring squares overlap, clusters can share points. This is the 1-skeleton of the nerve of the clusters. Software outputs only the graph: three clusters sharing a point would give a filled triangle in the nerve, but the graph keeps a hollow one.}
\begin{block}{Construction}
Each cluster is a vertex. Two clusters are joined by an edge whenever they share at least one point, which can happen because neighboring cubes overlap.
\src{singh2007mapper}
\end{block}
\begin{itemize}
  \item The clusters form a cover of (the non-noise part of) $P$, and the graph is the \alert{1-skeleton of its nerve}.
  \item Mapper software such as KeplerMapper outputs this graph only. Three clusters with a common point would form a triangle in the nerve, but the graph keeps just the three edges.
\end{itemize}
\end{frame}

\subsection{Checking a Mapper picture}
\begin{frame}{The nerve's loops are bounded, the graph's are not}
\note[item]{\textbf{Say:} Dey, M\'emoli, and Wang: for a nice cover of a nice space, the nerve's first homology is a quotient of the space's. So the nerve cannot have more loops than the space: $b_1$ of Mapper is at most $b_1$ of $X$. But that is the nerve. The graph drops the triangles, so it can gain loops. Three disks: the disk has no loop and neither does the nerve, but the graph does. We test this on our data at the end.}
\begin{block}{Theorem (Dey--M\'emoli--Wang)}
If $\mathcal U$ is a finite open cover of a compact, path-connected space $X$ by path-connected sets, the nerve map $\varphi : X \to |N(\mathcal U)|$ induces a surjection $H_1(X) \to H_1(|N(\mathcal U)|)$ ($\Z_2$ coefficients).
For a well-behaved $f$ (finitely many open path components per preimage), Mapper's pullback cover is such a cover, so
\[ b_1(\text{Mapper}) \le b_1(X). \]
\src{deymemoliwang2017}
\end{block}
\begin{itemize}\small
  \item The bound is about the nerve. The graph is only its 1-skeleton, so it can gain loops.
  \item Three disks with a common point (slide on nerves): the disk has $b_1 = 0$, the nerve is a filled triangle ($b_1 = 0$), but the graph is a hollow triangle ($b_1 = 1$).
  \item We test this on our data at the end, when we count the graph's loops.
\end{itemize}
\note[item]{The nerve map: $\varphi(x) = \sum_\alpha \rho_\alpha(x)\, u_\alpha$, with $\{\rho_\alpha\}$ a partition of unity subordinate to $\mathcal U$: each point goes to a weighted average of the vertices of the sets that contain it. $H_1$ of a space is singular homology, which agrees with simplicial homology on $|K|$ (Hatcher Thm.\ 2.27).}
\end{frame}

\begin{frame}{Checking a Mapper picture: the Rips filtration}
\note[item]{\textbf{Say:} A way to check a Mapper picture without Mapper: persistent homology. The Rips complex at scale $t$ fills in any set of points that are all within distance $t$ of each other. As $t$ grows, loops appear and then get filled in. Each inclusion gives a map on homology, so we can follow each loop's life.}
\centering
\includegraphics[width=0.66\textwidth, trim={0 32 140 0}, clip]{figures/rips_toy.pdf}
\raggedright
\begin{block}{Definition (Vietoris--Rips filtration)}
For a finite metric space $(P, \delta)$ and $t \ge 0$,
\[ \mathrm{Rips}_t(P) = \{\, \emptyset \ne S \subseteq P : \diam S \le t \,\}. \]
For $s \le t$, $\mathrm{Rips}_s(P) \subseteq \mathrm{Rips}_t(P)$, so inclusion induces linear maps $H_k(\mathrm{Rips}_s) \to H_k(\mathrm{Rips}_t)$.
\src{bauer2021ripser,zomorodiancarlsson2005}
\end{block}
{\small As $t$ grows, loops appear and are then filled in. Persistence records when.}
\note[item]{The loop in the figure dies at $d = 1.62$. The third panel is drawn at $t = 1.94$ because a Rips complex can fill a loop before its flat picture covers the center.}
\end{frame}

\begin{frame}{Checking a Mapper picture: persistence diagrams}
\note[item]{\textbf{Say:} Each loop is born at one scale and dies at a later one. Plot each loop as a point (birth, death): that is the persistence diagram. Points far from the diagonal live a long time and are real features; points near the diagonal are noise. In $H_0$ the classes are components, so a long-lived component is a gap between groups.}
\begin{columns}[c]
\begin{column}{0.36\textwidth}\centering
\includegraphics[width=\textwidth, trim={378 0 0 0}, clip]{figures/rips_toy.pdf}
\end{column}
\begin{column}{0.62\textwidth}
\begin{block}{Definitions}
\small
\begin{itemize}
  \item A class is \alert{born} at $b$ if it first appears in $H_k(\mathrm{Rips}_b)$, and \alert{dies} at $d$ when it maps to zero or merges with an older class.
  \item Its \alert{persistence} is $d - b$. The \alert{persistence diagram} is the multiset of pairs $(b, d)$.
  \item A \alert{persistent loop} is an $H_1$ point far from the diagonal; points near it are usually noise.
  \item In $H_0$ the classes are components, so a long $H_0$ bar is a gap.
\end{itemize}
\src{edelsbrunnerletscherzomorodian2002,fasy2014confidence}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Checking a Mapper picture: null models}
\note[item]{\textbf{Say:} A null model is fake data that keeps some features of the real data and destroys the rest. We run the same pipeline on it and compare. The same-covariance Gaussian is the blandest data with the same spread. The within-type shuffle shuffles each measurement among pitches of the same type, which breaks how the measurements go together. The rule: a feature counts only if the real data show more of it than every null. Not beating a null does not mean the feature is absent, only that covariance can explain it.}
\begin{block}{Idea}
A \alert{null model} is fake data that keeps some features of the real data and destroys the rest. We run the identical pipeline on it and compare.
\src{blumberg2014,robinsonturner2017,topaz2026null}
\end{block}
\begin{itemize}\small
  \item \alert{Same-covariance Gaussian}: keeps the mean and covariance, nothing else. Among distributions with these moments it has the least extra structure.
  \item \alert{Within-type shuffle}: permutes each coordinate among shapes of the same pitch type. Keeps each type's ranges, destroys how the measurements go together.
\end{itemize}
\begin{block}{Rule}
A feature counts as structure only if the real data show \alert{more} of it than every replicate of every null. Not beating a null does not prove the feature is absent, only that covariance can explain it.
\end{block}
\note[item]{Flight space (backup) uses a third null, the spin-pair shuffle: the same, but $(\cos\theta, \sin\theta)$ move together.}
\end{frame}

\end{enumerate}

\begin{frame}{Backup: Reeb graphs}
\note[item]{\textbf{Say:} If asked what Mapper approximates: the Reeb graph collapses each connected piece of each level set to a point. On a standing torus with height it is a circle with two whiskers.}
\note[item]{A multigraph is not a simplicial complex as defined in part 1; subdividing its edges makes it a simple graph, hence a complex.}
\begin{block}{Definition (Reeb graph)}
Let $f : X \to \R$ be continuous. Write $x \sim y$ if $f(x) = f(y)$ and $x, y$ lie in the same connected component of the level set $f^{-1}(f(x))$. The \alert{Reeb graph} is the quotient space $R_f(X) = X/\!\sim$.
\src{carriereoudot2018,reeb1946}
\end{block}
If $X$ is a compact manifold and $f$ is \alert{Morse} (smooth, with invertible Hessian at every critical point \parencite[Def.~1.21]{debray2018morse}), or more generally of \emph{Morse type} (backup), then $R_f(X)$ is a graph, technically a multigraph \parencite[p.~1334]{carriereoudot2018}.

\begin{column}{0.46\textwidth}\centering
\includegraphics[width=0.82\textwidth]{figures/reeb_torus.pdf}
\end{column}
\begin{column}{0.5\textwidth}\small
Standing torus, $f$ = height: Morse, with a minimum, two saddles and a maximum. Each level set collapses to one point per component: one near the top and bottom, two in the middle band. The Reeb graph is a circle with two whiskers.
\end{column}
\end{columns}

\begin{frame}{Backup: functions of Morse type}
\note[item]{\textbf{Say:} The condition that makes the Reeb graph a finite graph: finitely many critical values, and the space looks like a product between them.}
\begin{block}{Definition}
A continuous $f : X \to \R$ on a compact space $X$ is of \alert{Morse type} if
\begin{enumerate}
  \item it has finitely many critical values $a_1 < \dots < a_n$, and over each open interval between them (with $a_0 = -\infty$, $a_{n+1} = +\infty$) there are a compact, locally connected space $Y_i$ and a homeomorphism $\mu_i : Y_i \times (a_i, a_{i+1}) \to f^{-1}(a_i, a_{i+1})$ with $f \circ \mu_i$ the projection onto the second factor;
  \item each $\mu_i$ extends continuously to the closed interval;
  \item every level set has finitely generated homology.
\end{enumerate}
\par\vspace{2pt}\hfill{\scriptsize\textcolor{gray}{\parencite[Def.~1]{carriereoudot2018}}}
\end{block}
Morse functions are of Morse type, but not conversely: a Morse-type function need not be differentiable, and $X$ need not be a manifold. \hfill\parencite{carriereoudot2018}
\end{frame}

\begin{frame}{Backup: the DBSCAN algorithm}
\note[item]{\textbf{Say:} Start at any unvisited point; if it is a core point, grow a cluster from it, otherwise mark it noise for now. Border points marked noise can be picked up later.}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{algorithmic}[1]
\small
\State all points start \textsc{unclassified}
\For{each point $p$}
  \If{$p$ is \textsc{unclassified}}
    \If{\Call{ExpandCluster}{$p$}}
      \State start a new cluster id
    \EndIf
  \EndIf
\EndFor
\end{algorithmic}
\vspace{2pt}
{\scriptsize\textcolor{gray}{Main loop adapted from \textcite[p.~229]{ester1996dbscan}.}}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{ExpandCluster($p$):}
\begin{itemize}\small
  \item If $p$ is not a core point, mark it noise for now and return false.
  \item Otherwise give every unclassified or noise point of $N_\varepsilon(p)$ the current cluster id, and repeat from each new core point found.
  \item A border point first marked noise is relabeled if a later cluster reaches it.
\end{itemize}
\end{column}
\end{columns}
\note[item]{Walk through the previous slide's example with the pseudocode: start at a core point, expand, reach the border point, leave the crosses as noise.}
\end{frame}

\begin{frame}{Backup: why DBSCAN is correct}
\note[item]{\textbf{Say:} A cluster is determined by any one of its core points, so the algorithm finds every cluster no matter where it starts. The only order dependence is a border point shared by two clusters.}
\begin{block}{Lemma}
If $C$ is a cluster and $p \in C$ is a core point, then $C = \{\, o \in P : o \text{ is density-reachable from } p \,\}$.
\src{ester1996dbscan}
\end{block}
\begin{itemize}
  \item A cluster is determined by \emph{any one} of its core points. ExpandCluster collects everything density-reachable from $p$, so started at a core point it finds the cluster of $p$.
  \item Started at a non-core point it returns false, and nothing is lost: a point in a cluster is reached from that cluster's core points, and a point in no cluster stays noise.
  \item One exception: a border point in two clusters keeps the id of the one found first. Otherwise the result does not depend on the visit order. \hfill\parencite[pp.~229--230]{ester1996dbscan}
  \item Although the paper's Definition 6 allows different $(\varepsilon_i, M_i)$ for each cluster, the algorithm uses \alert{one global pair}, and so do we. \hfill\parencite{ester1996dbscan}
  \item Ours: $\varepsilon = 1.0$ (in standardized units) and $M = 4$.
\end{itemize}
\end{frame}

\begin{frame}{Backup: topological spaces}
\note[item]{\textbf{Say:} If asked for the formal setup: a topology is the collection of open sets. For us everything lives in $\R^n$ with the usual distance.}
\begin{block}{Definition (topology)}
A \alert{topology} on a set $X$ is a collection $\tau$ of subsets of $X$ (the \emph{open sets}) such that
\begin{enumerate}
  \item $\emptyset \in \tau$ and $X \in \tau$;
  \item any union of sets in $\tau$ is in $\tau$;
  \item any finite intersection of sets in $\tau$ is in $\tau$.
\end{enumerate}
\src{munkres2000topology}
\end{block}
\begin{block}{Definition (subspace topology)}
For $Y \subseteq X$, the \alert{subspace topology} is $\tau_Y = \{\, Y \cap U : U \in \tau \,\}$.
\src{munkres2000topology}
\end{block}
For us, $X$ is always a subset of $\R^n$ with the topology coming from Euclidean distance.
\end{frame}
\begin{frame}{Backup: homotopy}
\note[item]{\textbf{Say:} A homotopy is a continuous deformation from one map to another. A disk is contractible, an annulus is equivalent to a circle, and a circle is not contractible.}
\begin{block}{Definition (homotopy)}
Continuous maps $f, g : X \to Y$ are \alert{homotopic}, $f \simeq g$, if there is a continuous $F : X \times [0,1] \to Y$ with $F(x,0) = f(x)$ and $F(x,1) = g(x)$ for all $x \in X$.
\src{hatcher2001}
\end{block}
\begin{block}{Definition (homotopy equivalence)}
$X$ and $Y$ are \alert{homotopy equivalent}, $X \simeq Y$, if there are maps $f : X \to Y$, $g : Y \to X$ with $g f \simeq \mathrm{id}_X$ and $f g \simeq \mathrm{id}_Y$.
A space homotopy equivalent to a point is \alert{contractible}.
\src{hatcher2001}
\end{block}
Examples: a disk is contractible; an annulus $\simeq$ a circle; a circle is \emph{not} contractible.
\end{frame}
