\section{Introduction}

\begin{frame}{A pitch is a point in a 9-dimensional space}
\begin{itemize}
  \item Every MLB pitch is tracked: velocity, movement, spin, release point, \ldots \hfill\parencite{savantcsvdocs}
  \item Each pitch also gets a \alert{label}: four-seam (FF), sinker (SI), cutter (FC), slider (SL), sweeper (ST), curveball (CU), changeup (CH), \ldots \hfill\parencite{mlbglossary_pitchtypes}
  \item The labels change: in 2023 Statcast added \alert{sweeper} and \alert{slurve} as new classifications \hfill\parencite{mlbglossary_sweeper,mlbglossary_slurve}
\end{itemize}
\vspace{6pt}
\begin{block}{Question}
Is the space of pitch shapes made of separate clumps (so the labels are natural), or a \alert{continuum} that the labels cut into pieces?
\end{block}
\vspace{4pt}
We answer with the \alert{Mapper algorithm}, a tool from topological data analysis, and we test every feature it shows us.
\end{frame}

\begin{frame}{Roadmap}
\begin{enumerate}
  \item \textbf{Topological foundations}: simplicial complexes, graphs as complexes, nerves, homology, Reeb graphs
  \item \textbf{The Mapper algorithm}: lens (PCA), cover, clustering (DBSCAN), graph
  \item \textbf{MLB pitch data}: what Mapper shows, and how we check it is real
\end{enumerate}
\vspace{8pt}
\textbf{Thread to watch:} the \emph{nerve} of a cover appears in part 1 as a theorem, in part 2 as the last step of the algorithm, and in part 3 as the reason the graph's loops are not real.
\end{frame}

\section{Topological Foundations}

\begin{frame}{Simplices}
\begin{block}{Definition ($n$-simplex)}
Points $v_0,\dots,v_n \in \R^m$ are \alert{affinely independent} if $v_1 - v_0, \dots, v_n - v_0$ are linearly independent. Their convex hull
\[
[v_0,\dots,v_n] = \Big\{ \textstyle\sum_{i=0}^n t_i v_i \;:\; t_i \ge 0,\ \sum_i t_i = 1 \Big\}
\]
is an \alert{$n$-simplex} with \emph{vertices} $v_i$. The simplex spanned by any nonempty subset of the vertices is a \alert{face}.
\src{hatcher2001}
\end{block}

\begin{frame}{Simplicial complexes}
\begin{block}{Definition (simplicial complex)}
A \alert{simplicial complex} $K$ is a finite collection of simplices such that
\begin{enumerate}
  \item every face of a simplex of $K$ is in $K$;
  \item any two simplices of $K$ are disjoint or intersect in a common face.
\end{enumerate}
Its \alert{vertex set} is the set of its 0-simplices.
\src{edelsbrunnerletscherzomorodian2002}
\end{block}
\begin{block}{Abstract version}
An \alert{abstract simplicial complex} on a vertex set $V$ is a collection $S$ of finite nonempty subsets of $V$ with $\{v\} \in S$ for all $v$, and $\tau \subseteq \sigma \in S \Rightarrow \tau \in S$. A $k$-simplex is a set of $k+1$ vertices. Placing vertices in general position gives a geometric realization $|K|$.
\src{zomorodiancarlsson2005}
\end{block}
\end{frame}

\begin{frame}{Graphs are simplicial complexes}
\begin{block}{Definition ($k$-skeleton)}
The \alert{$k$-skeleton} $K^{(k)}$ of a simplicial complex $K$ is the subcomplex of all simplices of dimension $\le k$; the 0-skeleton is the vertex set.
\src{edelsbrunnerharerpatel2008}
\end{block}
By the abstract definition, a finite graph $G = (V, E)$ \emph{is} a simplicial complex of dimension $\le 1$ (vertices = 0-simplices, edges = 1-simplices); its realization $|G|$ is a \alert{topological graph}.\par

\end{itemize}

\begin{frame}{The nerve of a cover}
\begin{block}{Definition (nerve)}
Let $\mathcal U = \{U_\alpha\}_{\alpha \in A}$ be a finite cover of a space $X$. The \alert{nerve} $N(\mathcal U)$ is the abstract simplicial complex with vertex set $A$, where a set $\{\alpha_0,\dots,\alpha_k\} \subseteq A$ spans a $k$-simplex if and only if
\[ U_{\alpha_0} \cap \cdots \cap U_{\alpha_k} \ne \emptyset. \]
\src{singh2007mapper,hatcher2001}
\end{block}

\begin{frame}{The Nerve Theorem}
\begin{block}{Definitions}
$C \subseteq \R^d$ is \alert{convex} if the segment between any two points of $C$ lies in $C$.
Spaces $X \simeq Y$ are \alert{homotopy equivalent} if each can be continuously deformed into the other (precise definition in backup); \emph{contractible} means $\simeq$ a point. E.g.\ annulus $\simeq$ circle, disk $\simeq$ point.
\src{hatcher2001}
\end{block}
\begin{block}{Theorem (Nerve Theorem, closed convex version; cited, not proved here)}
If $X \subseteq \R^d$ is covered by finitely many closed convex sets $\mathcal C = \{C_1,\dots,C_n\}$, then $X \simeq |N(\mathcal C)|$.
\par\vspace{2pt}\hfill{\scriptsize\textcolor{gray}{\parencite[Thm.~3.9]{bauerkerberrollrolle2023}; an early closed-cover version: \parencite[Cor.~3]{borsuk1948}}}
\end{block}
\begin{itemize}\small
  \item Why it works: intersections of convex sets are convex, hence empty or contractible.
  \item Open-cover version: if $\mathcal U$ is an open cover of a paracompact $X$ and every nonempty finite intersection is contractible, then $X \simeq N(\mathcal U)$. \hfill\parencite{hatcher2001}
  \item \alert{Warning for later:} covers in data analysis are rarely this nice, so a nerve need not have the shape of the space.
\end{itemize}
\end{frame}

\begin{frame}{Counting holes: Betti numbers}
\begin{block}{Definition}
The $k$-th \alert{Betti number} $b_k(X)$ is the rank of the $k$-th homology group $H_k(X)$. We work with $\Z_2$ coefficients.
\src{hatcher2001}
\end{block}
\begin{itemize}
  \item $b_0$ = number of path components \parencite{hatcher2001}; $b_1$ = number of independent loops.
  \item For a graph $G$ (a 1-dimensional complex) with $V$ vertices, $E$ edges, $C$ components, the Euler characteristic gives $V - E = b_0 - b_1$, so
  \[ b_1(G) = E - V + C. \]
  \item For a 2-dimensional complex, triangles can \emph{fill} loops: $b_1 = E - V + C - \rank \partial_2$, where $\partial_2$ maps each triangle to its boundary edges.
\end{itemize}
\vspace{4pt}
Hollow triangle: $3 - 3 + 1 = 1$ loop. \quad Filled triangle: $1 - 1 = 0$ loops.
\end{frame}


\begin{frame}{Reeb graphs}
\begin{block}{Definition (Reeb graph)}
Let $f : X \to \R$ be continuous. Write $x \sim y$ if $f(x) = f(y)$ and $x, y$ lie in the same connected component of the level set $f^{-1}(f(x))$. The \alert{Reeb graph} is the quotient space $R_f(X) = X/\!\sim$.
\src{carriereoudot2018,reeb1946}
\end{block}
When $X$ is a smooth manifold and $f$ is a Morse function (more generally, of Morse type), $R_f(X)$ is a graph, technically a multigraph: a \alert{topological graph}. \hfill\parencite[p.~1334]{carriereoudot2018}
\begin{columns}[c]
\begin{column}{0.55\textwidth}\centering
\includegraphics[width=0.66\textwidth]{figures/reeb_torus.pdf}
\end{column}
\begin{column}{0.4\textwidth}\small
Torus, $f$ = height. Each level set collapses to one point per component: one near the top and bottom, two in the middle band. The Reeb graph is a circle with two whiskers.
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Mapper: the topological version}
\begin{block}{Definition (Mapper)}
Let $f : X \to Z$ be continuous and $\mathcal U = \{U_\alpha\}$ a finite open cover of $Z$. Split each preimage $f^{-1}(U_\alpha)$ into its \alert{path components}; these form the \alert{pullback cover} $f^*\mathcal U$ of $X$. The \alert{Mapper} is its nerve, $M(\mathcal U, f) = N(f^*\mathcal U)$.
\src{singh2007mapper,deymemoliwang2017}
\end{block}
\begin{itemize}\small
  \item Cited result: for $Z = \R$, Mapper is a discrete estimator of the Reeb graph; for $Z = \R^d$ it approximates the analogous \emph{Reeb space}. \hfill\parencite{carrieremicheloudot2018,munchwang2016}
\end{itemize}
\end{frame}

\begin{frame}{From spaces to data}
\begin{block}{Definitions}
\begin{itemize}
  \item A \alert{point cloud} is a finite set $P \subseteq \R^n$ with Euclidean distance.
  \item In the \emph{statistical} version of Mapper, clustering plays the role of path components: each preimage $f^{-1}(U_\alpha) \cap P$ is split into \alert{clusters}. \hfill\parencite{singh2007mapper}
  \item The \alert{Mapper graph} is the 1-skeleton of the resulting nerve: one vertex per cluster, one edge per pair of clusters that share a point.
\end{itemize}
\end{block}
\vspace{4pt}
The 1-skeleton keeps vertices and edges but \alert{drops the triangles}. Remember this: it is why the graph's loops will turn out not to be real.
\end{frame}

\begin{frame}{Overview}
\centering

  \node[box] (a) {point cloud\\$P \subseteq \R^n$};
  \node[box, right=of a] (b) {\textbf{1. lens}\\$f : P \to \R^k$\\(PCA)};

\vspace{10pt}
\begin{itemize}
  \item Steps 1--3 build a cover of $P$; step 4 takes its nerve. \hfill\parencite{singh2007mapper}
  \item Clustering happens in the full space $\R^n$; the lens only decides which points are clustered together.
\end{itemize}
\end{frame}

\subsection{Step 1: the lens}
\begin{frame}{Step 1: choosing a lens}
\begin{block}{Lens (filter) function}
A continuous map $f : P \to \R^k$, usually with $k$ small, used to organize the cover.
\src{singh2007mapper}
\end{block}
\begin{itemize}
  \item Among the Mapper applications reviewed by \textcite{madukpe2026review}, \alert{PCA} is the most commonly used lens. We use a 2-dimensional PCA lens.
  \item Other lenses (velocity $\times$ vertical movement, Isomap, a density lens) are used later as a robustness check. \hfill\parencite{madukpe2026review,tenenbaum2000isomap}
\end{itemize}
\end{frame}

\begin{frame}{PCA: goal and setup}
\begin{block}{Goal}
Find new uncorrelated variables, linear combinations of the old ones, that successively capture as much variance as possible.
\src{jolliffecadima2016,pearson1901}
\end{block}
\begin{block}{Setup: why we center}
Data: $m$ observations of $n$ features. Subtract each column's mean to get the centered matrix $\mathbf A \in \R^{m \times n}$.
\begin{itemize}
  \item Variance is measured around the mean, so centering does not change the covariance matrix or the principal components.
  \item It makes $(m-1)\,S = \mathbf A^{\mathsf T}\mathbf A$, which turns PCA into a statement about $\mathbf A$ itself, solvable by the SVD.
\end{itemize}
\src{jolliffecadima2016}
\end{block}
\end{frame}

\begin{frame}{The covariance matrix}
\begin{block}{Definition}
$S = \dfrac{1}{m-1}\,\mathbf A^{\mathsf T}\mathbf A \in \R^{n \times n}$: diagonal entries are the feature variances, off-diagonal entries the covariances.
\end{block}
\begin{block}{Fact: $S$ is positive semidefinite}
For any $x \in \R^n$: \quad $x^{\mathsf T} S x = \dfrac{1}{m-1}\, x^{\mathsf T}\mathbf A^{\mathsf T}\mathbf A x = \dfrac{1}{m-1}\,\|\mathbf A x\|^2 \ \ge 0.$
\end{block}
So $S$ has real, nonnegative eigenvalues and an orthonormal basis of eigenvectors. The variance of the data along a unit vector $v$ is $v^{\mathsf T} S v$, maximized by the top eigenvector.
\end{frame}

\begin{frame}{Singular value decomposition}
\begin{block}{Theorem (SVD, matrix version)}
Let $\mathbf A$ have rank $r \ge 1$. Then $\mathbf A = \mathbf U \boldsymbol\Sigma \mathbf V^{\mathsf T}$, where $\mathbf U \in \R^{m \times r}$ and $\mathbf V \in \R^{n \times r}$ have orthonormal columns and $\boldsymbol\Sigma = \mathrm{diag}(\sigma_1 \ge \cdots \ge \sigma_r > 0)$.
\src{axler2024ladr}
\end{block}
\begin{itemize}
  \item The columns of $\mathbf V$ (right singular vectors) are eigenvectors of $\mathbf A^{\mathsf T}\mathbf A$, and the columns of $\mathbf U$ are eigenvectors of $\mathbf A\mathbf A^{\mathsf T}$, both for the nonzero eigenvalues $\sigma_i^2$. \hfill\parencite{jolliffecadima2016}
  \item Hence the eigenvalues of $S$ are $\lambda_i = \sigma_i^2/(m-1)$, and the principal directions are the columns $v_i$ of $\mathbf V$. \hfill\parencite{jolliffecadima2016}
  \item The largest variance is $\lambda_1 = \sigma_1^2/(m-1)$.
\end{itemize}
\end{frame}

\begin{frame}{Projecting to $k$ dimensions}
\begin{itemize}
  \item Keep the first $k$ directions: $\mathbf V_k = [v_1 \cdots v_k] \in \R^{n \times k}$.
  \item The lens is $f(x) = \mathbf V_k^{\mathsf T} x$; the score matrix is $\mathbf Z_k = \mathbf A \mathbf V_k = \mathbf U_k \boldsymbol\Sigma_k$.
\end{itemize}
\begin{block}{Best rank-$k$ approximation}
The matrix $\mathbf A_k = \mathbf A\mathbf V_k\mathbf V_k^{\mathsf T} = \mathbf U_k\boldsymbol\Sigma_k\mathbf V_k^{\mathsf T}$ is the closest rank-$\le k$ matrix to $\mathbf A$:
in the operator norm, with error $\sigma_{k+1}$ \parencite{axler2024ladr}, and in the least-squares sense \parencite{jolliffecadima2016}.
\end{block}
\begin{block}{Variance explained}
$\mathrm{PVE}_j = \lambda_j / \sum_l \lambda_l = \sigma_j^2 / \sum_l \sigma_l^2$, \qquad $\mathrm{CVE}_k = \sum_{j \le k}\mathrm{PVE}_j$.
\src{jolliffecadima2016}
\end{block}
\end{frame}

\begin{frame}{Why $k = 2$}
\begin{itemize}
  \item On our data, PC1 explains \alert{38.5\%} of the variance and PC2 \alert{21.0\%} (59.5\% together).
  \item The cover lives in $\R^k$: with 10 intervals per axis, $k=2$ gives 100 cubes, while $k = 3$ would give 1{,}000, most nearly empty.
  \item A 2-D lens can be drawn and read, which matters for interpreting the result.
  \item PCA is a \alert{linear} projection, so curved structure can be flattened. That is acceptable here because clustering still happens in the full 9-dimensional space, and we re-run everything with three other lenses, including the nonlinear Isomap.
\end{itemize}
\end{frame}

\subsection{Step 2: the cover}
\begin{frame}{Step 2: building a cover}
\begin{itemize}
  \item Cover the image $f(P) \subseteq \R^k$ by overlapping hypercubes: each axis is cut into intervals that overlap their neighbors by a fixed percentage. \hfill\parencite{singh2007mapper,madukpe2026review}
  \item Two parameters control the resolution: the \alert{number of intervals} and the \alert{percentage overlap}. \hfill\parencite{singh2007mapper}
  \item The resulting graph depends on these choices, and tuning them on real data is not easy. \hfill\parencite{carrieremicheloudot2018}
\end{itemize}
\vspace{6pt}

\end{tikzpicture}
\end{frame}

\subsection{Step 3: clustering with DBSCAN}
\begin{frame}{Step 3: clustering each preimage}
\begin{itemize}
  \item For each cube $U_i$, cluster the points of $f^{-1}(U_i)$ in the full space $\R^n$. Clusters stand in for path components. \hfill\parencite{singh2007mapper}
  \item We use \alert{DBSCAN}, the second most common choice in reviewed Mapper applications (after hierarchical clustering). \hfill\parencite{madukpe2026review}
  \item Why DBSCAN: it needs no preset number of clusters, finds clusters of any shape, and labels isolated points as \alert{noise}. \hfill\parencite{ester1996dbscan}
\end{itemize}
\vspace{6pt}
\textbf{Setting for the next slides:} a point cloud $P$ with distance $d$, a radius $\varepsilon > 0$, and a count $M \in \N$ (called Eps and MinPts in the original paper).
\end{frame}

\begin{frame}{Density reachability}
\begin{block}{Definition 1 ($\varepsilon$-neighborhood)}
$N_\varepsilon(p) = \{\, q \in P : d(p,q) \le \varepsilon \,\}$. \quad A point $p$ is a \alert{core point} if $|N_\varepsilon(p)| \ge M$.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 2 (directly density-reachable)}
$p$ is directly density-reachable from $q$ if $p \in N_\varepsilon(q)$ and $q$ is a core point.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 3 (density-reachable)}
$p$ is density-reachable from $q$ if there is a chain $q = p_1, p_2, \dots, p_k = p$ with each $p_{i+1}$ directly density-reachable from $p_i$.
\src{ester1996dbscan}
\end{block}
\end{frame}

\begin{frame}{Clusters and noise}
\begin{block}{Definition 4 (density-connected)}
$p$ and $q$ are density-connected if both are density-reachable from some common point $o$.
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 5 (cluster)}
A \alert{cluster} is a nonempty $C \subseteq P$ such that
(i) if $p \in C$ and $q$ is density-reachable from $p$, then $q \in C$ (\emph{maximality}), and
(ii) any two points of $C$ are density-connected (\emph{connectivity}).
\src{ester1996dbscan}
\end{block}
\begin{block}{Definition 6 (noise)}
\alert{Noise} is the set of points of $P$ in no cluster. It depends on the chosen $(\varepsilon, M)$.
\src{ester1996dbscan}
\end{block}
\end{frame}

\begin{frame}{Example: core, border, noise}
\centering
\includegraphics[width=0.72\textwidth]{figures/dbscan_toy.pdf}
\begin{itemize}\small
  \item Solid circle: a \alert{core point}; its $\varepsilon$-ball holds 6 points $\ge M = 4$.
  \item Dashed circle: a \alert{border point}; only 3 points in its ball, but a core point is among them, so it is density-reachable and joins the cluster.
  \item Crosses: \alert{noise}; each is more than $\varepsilon$ from every core point. With $\varepsilon = 0.7$ some would join a cluster: noise depends on $(\varepsilon, M)$.
\end{itemize}
\end{frame}

\begin{frame}{Why the algorithm is correct}
\begin{block}{Lemma}
If $C$ is a cluster and $p \in C$ is a core point, then $C = \{\, o \in P : o \text{ is density-reachable from } p \,\}$.
\src{ester1996dbscan}
\end{block}
\begin{itemize}
  \item So a cluster is determined by \emph{any one} of its core points: start anywhere inside and collect everything reachable.
  \item The paper's Definition 6 allows different $(\varepsilon_i, M_i)$ for each cluster; the algorithm, and our analysis, use \alert{one global pair}. \hfill\parencite{ester1996dbscan}
  \item Ours: $\varepsilon = 1.0$ (in standardized units) and $M = 4$.
\end{itemize}
\end{frame}

\begin{frame}{The DBSCAN algorithm}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{algorithmic}[1]
\small
\State all points start \textsc{unclassified}
\For{each point $p$}
  \If{$p$ is \textsc{unclassified}}
    \If{\Call{ExpandCluster}{$p$}}
      \State start a new cluster id
    \EndIf
  \EndIf
\EndFor
\end{algorithmic}
\vspace{2pt}
{\scriptsize\textcolor{gray}{Main loop adapted from \textcite[p.~229]{ester1996dbscan}.}}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{ExpandCluster($p$):}
\begin{itemize}\small
  \item If $p$ is not a core point, mark it noise for now and return false.
  \item Otherwise give every point of $N_\varepsilon(p)$ the current cluster id, and repeat from each new core point found.
  \item By the lemma, this collects exactly the cluster of $p$.
  \item A border point first marked noise is relabeled if a later cluster reaches it.
\end{itemize}
\end{column}
\end{columns}
\note[item]{Walk through the previous slide's example with the pseudocode: start at a core point, expand, reach the border point, leave the crosses as noise.}
\end{frame}

\subsection{Step 4: the graph}
\begin{frame}{Step 4: building the graph}
\begin{block}{Construction}
Each cluster is a vertex. Two clusters are joined by an edge whenever they share at least one point, which can happen because neighboring cubes overlap.
\src{singh2007mapper}
\end{block}
\begin{itemize}
  \item The clusters form a cover of (the non-noise part of) $P$, and the graph is the \alert{1-skeleton of its nerve}.
  \item Software outputs this graph only. Three clusters with a common point would form a triangle in the nerve, but the graph keeps just the three edges.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{KeplerMapper}
\begin{itemize}
  \item Python implementation of Mapper; works with scikit-learn-compatible clustering and scaling algorithms. \hfill\parencite{vanveen2019keplermapper,pedregosa2011sklearn}
  \item Developed under the scikit-tda project. \hfill\parencite{keplermapper_repo}
\end{itemize}
\begin{lstlisting}
import kmapper as km
from sklearn.cluster import DBSCAN
from sklearn.decomposition import PCA

mapper = km.KeplerMapper()
lens = mapper.fit_transform(X, projection=PCA(n_components=2))
graph = mapper.map(lens, X,
                   cover=km.Cover(n_cubes=10, perc_overlap=0.3),
                   clusterer=DBSCAN(eps=1.0, min_samples=4))
\end{lstlisting}
\end{frame}

\begin{frame}{The graph has 86 loops. Are they real?}
\centering
\includegraphics[width=0.58\textwidth]{figures/nerve_b1.pdf}
\par\small
Graph: $b_1 = E - V + C = 143 - 66 + 9 = 86$. The full nerve adds 113 triangles; $\partial_2$ is a $143 \times 113$ matrix over $\Z_2$ of rank 85, so $b_1 = 86 - 85 = 1$. Across 27 parameter choices, graph $b_1$ ranges 33--147; nerve $b_1$ stays 0--4, in the same range as the null models.
\end{frame}

\begin{frame}{Why: nerves can only kill loops}
\begin{block}{Theorem (Dey--M\'emoli--Wang; cited, not proved here)}
If $\mathcal U$ is a path-connected cover of $X$, the nerve map induces a surjection $H_1(X) \to H_1(N(\mathcal U))$ ($\Z_2$ coefficients, $X$ compact).
Mapper's pullback cover is path-connected by construction, so
\[ b_1(\text{Mapper}) \le b_1(X). \]
\src{deymemoliwang2017}
\end{block}
\begin{itemize}\small
  \item Persistent homology of the 3{,}943 pitch shapes shows \alert{no persistent loops}, just like a Gaussian null (backup). So the underlying space has $b_1 \approx 0$, and any real Mapper of it must too.
  \item The 86 graph loops come from reading only the 1-skeleton: triangles fill them.
  \item Honest scope: the theorem is about a space; we have a finite sample. Theorem plus consistent evidence, not a proof about the data.
\end{itemize}
\end{frame}



\begin{frame}{Labels cut a continuum}
\begin{itemize}
  \item One pitcher\'s two labels most often share or neighbor Mapper nodes for FF/SI, CH/FF, CH/SI, FC/FF, FC/SI (raw counts, which also reflect how often pitchers throw both).
  \item About \alert{55\%} of individual pitches fall in two or more overlapping Mapper nodes (opening week of 2025; the same in a week of April 2026).
  \item Statcast itself relabeled some of our pitch shapes between our fit and a later re-pull (SL$\leftrightarrow$ST, SI$\leftrightarrow$FF, ST/FC).
  \item Statcast added the sweeper as its own pitch type in 2023 because of its distinct shape and movement. \hfill\parencite{mlbglossary_sweeper}
\end{itemize}
\vspace{6pt}
\begin{block}{And for outcomes}
Statcast's labels explain \emph{more} outcome variance than our clusters (e.g.\ whiff rate: $\omega^2 = 2.8\%$ for labels vs.\ 1.5--1.9\% for clusters). Mapper's value here is describing shape, not predicting results.
\end{block}
\end{frame}

\begin{frame}{Why a continuum matters}
\textbf{Interpretation, not a tested result:}
\begin{itemize}\small
  \item \alert{A label is a region, not a kind of pitch.} Comparisons by pitch type can mislead near boundaries, and most pitches sit near one.
  \item \alert{Describe a pitch by where it sits}: its position in shape space and its nearest neighbors, not only its tag.
  \item \alert{Pitch design is movement through a connected space}: with no gaps, nearby shapes are realistic targets (next slide).
  \item \alert{For TDA practice:} check graph features against null models before reading meaning into them.
\end{itemize}
\begin{block}{Illustration (one pitcher, 2025)}
Brady Singer's slider (809 pitches) breaks 4.9 in horizontally; his ``sweeper'' (316 pitches) breaks 9.0 in. The average slider breaks about 6 in and the average sweeper about 15 in \parencite{mlbglossary_sweeper}: his sweeper sits between the two, closer to a slider.
\end{block}
\end{frame}

\begin{frame}{Conclusions}
\begin{block}{About pitches}
Pitch-shape space is a \alert{connected continuum} (about 6 of 9 dimensions; backup), with no loops and no hidden gaps beyond a Gaussian, plus a tail of very slow pitches. Pitch-type labels are cuts through it, and every pitch has real neighbors to compare against.
\end{block}
\begin{block}{About the method}
\begin{itemize}
  \item Mapper gives a readable picture; \alert{null models, homology, and dimension estimates certify it}.
  \item Read graph loops through the full nerve: the graph is only its 1-skeleton, and by the Dey--M\'emoli--Wang theorem a Mapper cannot have more loops than the space.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Future work}
\begin{itemize}

  \item \textbf{Arm slot.} Model movement \emph{relative to} a pitcher's arm angle instead of raw movement. \hfill\parencite{savantcsvdocs}
  \item \textbf{Replication.} Repeat on other seasons, and require a minimum number of pitches per shape.
\end{itemize}
\end{frame}

\begin{frame}{Persistent homology}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{block}{Vietoris--Rips complex}
For a finite metric space $(P, d)$ and scale $t \in \R$: $\mathrm{Rips}_t(P) = \{\, \emptyset \ne S \subseteq P : \diam S \le t \,\}$.
\src{bauer2021ripser}
\end{block}
\begin{itemize}\small
  \item As $t$ grows we get nested complexes; persistent homology tracks each hole from the scale where it is \emph{born} to where it \emph{dies}. \hfill\parencite{edelsbrunnerletscherzomorodian2002,zomorodiancarlsson2005}
  \item A \alert{persistence diagram} plots (birth, death); points far from the diagonal are real features, points near it are noise. \hfill\parencite{fasy2014confidence}
\end{itemize}
\end{column}

\begin{frame}{Backup: checks on the pitch-design example}
\begin{itemize}\small
  \item \alert{Mapper is not the filter.} The physical limits alone pick the same 10 neighbors (on average 99\% of them lie in the Mapper neighborhood). The topology's role is the continuum result: nearby shapes exist, so comparing to them is interpolation, not extrapolation.
  \item \alert{Tolerances.} Over 27 settings (velocity 1 / 1.5 / 2 mph $\times$ spin 150 / 200 / 300 rpm $\times$ movement 3 / 4 / 5 in), the pitch has a neighborhood in 24 and is the lowest-scoring in it at the median setting; better neighbors sit $+1.5$ to $+3.2$ in higher in induced vertical break.
  \item \alert{Model-free check.} Across 420 four-seamers (2025, $\ge 150$ pitches), holding velocity, horizontal break, spin, extension and release height fixed, swinging-strike rate rises 0.17 percentage points per inch of induced vertical break ($p = 0.02$). So $+2$ in is worth only $\approx 0.3$ points: most of his 5-point gap to the neighbors is something else (location, sequencing, deception, noise).
  \item \alert{Not every neighbor agrees:} the best one has \emph{less} ride but releases about 5 in lower.
  \item Model: CatBoost predictions of contact quality (xwOBA), swinging strike and chase from 9 physical features; 5-fold out-of-fold on 2{,}008{,}693 pitches (2022--24). Composite: equal-weight $z$-scores pooled across pitch types, 100 = average.
\end{itemize}

\begin{frame}{Backup: topological spaces}
\begin{block}{Definition (topology)}
A \alert{topology} on a set $X$ is a collection $\tau$ of subsets of $X$ (the \emph{open sets}) such that
\begin{enumerate}
  \item $\emptyset \in \tau$ and $X \in \tau$;
  \item any union of sets in $\tau$ is in $\tau$;
  \item any finite intersection of sets in $\tau$ is in $\tau$.
\end{enumerate}
\src{munkres2000topology}
\end{block}
\begin{block}{Definition (subspace topology)}
For $Y \subseteq X$, the \alert{subspace topology} is $\tau_Y = \{\, Y \cap U : U \in \tau \,\}$.
\src{munkres2000topology}
\end{block}
For us, $X$ is always a subset of $\R^n$ with the topology coming from Euclidean distance.
\end{frame}
\begin{frame}{Backup: homotopy}
\begin{block}{Definition (homotopy)}
Continuous maps $f, g : X \to Y$ are \alert{homotopic}, $f \simeq g$, if there is a continuous $F : X \times [0,1] \to Y$ with $F(x,0) = f(x)$ and $F(x,1) = g(x)$ for all $x \in X$.
\src{hatcher2001}
\end{block}
\begin{block}{Definition (homotopy equivalence)}
$X$ and $Y$ are \alert{homotopy equivalent}, $X \simeq Y$, if there are maps $f : X \to Y$, $g : Y \to X$ with $g f \simeq \mathrm{id}_X$ and $f g \simeq \mathrm{id}_Y$.
A space homotopy equivalent to a point is \alert{contractible}.
\src{hatcher2001}
\end{block}
Examples: a disk is contractible; an annulus $\simeq$ a circle; a circle is \emph{not} contractible.
\end{frame}
